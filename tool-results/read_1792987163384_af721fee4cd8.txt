    54→/** Resolved fix commands + files actively mentioned in the Knowledge Graph. */
    55→function extractFixesAndFiles(
    56→  items: BrainItem[],
    57→  edges: Array<{ sourceNode: string; relationType: string }>
    58→): { resolvedFixes: string[]; activeFiles: string[] } {
    59→  const resolvedFixes: string[] = [];
    60→  const activeFiles = new Set<string>();
    61→  for (const it of items) {
    62→    const fix = it.metadata?.fix;
    63→    if (typeof fix === "string" && fix) resolvedFixes.push(`${it.title} -> \`${fix}\``);
    64→  }
    65→  for (const e of edges) {
    66→    if (e.relationType === "MENTIONED_IN") activeFiles.add(e.sourceNode);
    67→  }
    68→  return { resolvedFixes, activeFiles: [...activeFiles] };
    69→}

    98→interface TimeBlock {
    99→  projectId: string | null;
   100→  items: Array<{ id: string; itemType: string; title: string; structuredSummary: string; rawContent: string; capturedAt: Date }>;
   101→  start: Date;

   105→/** Splits an ordered item stream into coding blocks whenever the gap exceeds 90 minutes. */
   106→function splitIntoBlocks(
   107→  rows: Array<{ id: string; itemType: string; title: string; structuredSummary: string; rawContent: string; projectId: string | null; capturedAt: Date }>
   108→): TimeBlock[] {
   109→  const blocks: TimeBlock[] = [];
   110→  let current: TimeBlock | null = null;
   111→
   112→  const groups = new Map<string | null, typeof rows>();
   113→  for (const row of rows) {
   114→    const key = row.projectId ?? "__inbox__";
   115→    const arr = groups.get(key) ?? [];
   116→    arr.push(row);
   117→    groups.set(key, arr);
   118→  }
   119→
   120→  for (const [key, groupRows] of groups) {
   121→    current = null;
   122→    for (const row of groupRows) {
   123→      if (
   124→        current &&
   125→        row.capturedAt.getTime() - current.end.getTime() <= SESSION_GAP_MS
   126→      ) {
   127→        current.items.push(row);
   128→        current.end = row.capturedAt;
   129→      } else {
   130→        current = {
   131→          projectId: key === "__inbox__" ? null : row.projectId,
   132→          items: [row],
   133→          start: row.capturedAt,
   134→          end: row.capturedAt,
   135→        };
   136→        blocks.push(current);
   137→      }
   138→    }
   139→  }
   140→  return blocks;
   141→}

   143→function fmtHM(d: Date): string {
   144→  return `${String(d.getUTCHours()).padStart(2, "0")}:${String(d.getUTCMinutes()).padStart(2, "0")}`;
   145→}

   147→async function rollupSessions(): Promise<CompressionReport["sessionDetails"]> {
   148→  const rows = await db.item.findMany({
   149→    where: { itemType: { not: "session" } }, // never re-rollup previous rollups
   150→    orderBy: { capturedAt: "asc" },
   151→    select: {
   152→      id: true,
   153→      itemType: true,
   154→      title: true,
   155→      structuredSummary: true,
   156→      rawContent: true,
   157→      projectId: true,
   158→      capturedAt: true,
   159→    },
   160→  });
   161→
   162→  const blocks = splitIntoBlocks(rows).filter(
   163→    (b) => b.items.length >= MIN_SESSION_ITEMS
   164→  );
   165→
   166→  const created: CompressionReport["sessionDetails"] = [];
   167→  for (const block of blocks) {
   168→    const first = block.items[0];
   169→    const last = block.items[block.items.length - 1];
   170→    // Deterministic, idempotent session key — same block never rolls up twice
   171→    const sessionKey = `sess_${block.projectId ?? "inbox"}_${first.id}_${last.id}`;
   172→
   173→    const existing = await db.item.findFirst({
   174→      where: { itemType: "session", metadata: { contains: sessionKey } },
   175→      select: { id: true },
   176→    });
   177→    if (existing) continue;
   178→
   179→    const typeCounts: Record<string, number> = {};
   180→    for (const it of block.items) {
   181→      typeCounts[it.itemType] = (typeCounts[it.itemType] ?? 0) + 1;
   182→    }
   183→    const typeStr = Object.entries(typeCounts)
   184→      .sort((a, b) => b[1] - a[1])
   185→      .map(([t, n]) => `${n} ${t}${n > 1 ? "s" : ""}`)
   186→      .join(", ");
   187→    const keywords = topKeywords(
   188→      block.items.map((i) => `${i.title} ${i.rawContent}`).join(" "),
   189→      6
   190→    );
   191→    const day = block.start.toISOString().slice(0, 10);
   192→    const projectName = block.projectId
   193→      ? (await db.project.findUnique({ where: { id: block.projectId }, select: { name: true } }))?.name ?? "Project"
   194→      : "Inbox";
   195→    const title = `Work Session ${day} · ${fmtHM(block.start)}–${fmtHM(last.capturedAt)} · ${block.items.length} captures`;
   196→
   197→    const rawContent =
   198→      `# ${title}\n\n` +
   199→      `**Project:** ${projectName}\n` +
   200→      `**Block:** ${block.start.toISOString()} → ${last.capturedAt.toISOString()} (gap-tolerant rollup, 90 min)\n` +
   201→      `**Members:** ${block.items.length} (${typeStr})\n` +
   202→      `**Top topics:** ${keywords.join(", ") || "—"}\n\n` +
   203→      `## Captures in this block\n` +
   204→      block.items
   205→        .map((i) => `- [${fmtHM(i.capturedAt)}] **${i.itemType}** ${i.title}${i.structuredSummary ? ` — ${i.structuredSummary.slice(0, 140)}` : ""}`)
   206→        .join("\n") +
   207→      "\n";
   208→
   209→    await saveItem({
   210→      title,
   211→      rawContent,
   212→      itemType: "session",
   213→      projectId: block.projectId,
   214→      structuredSummary: `Rolled up ${block.items.length} captures (${typeStr}) · topics: ${keywords.slice(0, 4).join(", ") || "—"}`,
   215→      sourceApp: "MemoryCompressionDaemon",
   216→      metadata: {
   217→        session_key: sessionKey,
   218→        session_start: block.start.toISOString(),
   219→        session_end: last.capturedAt.toISOString(),
   220→        member_ids: block.items.map((i) => i.id),
   221→        type_counts: typeCounts,
   222→        keywords,
   223→      },
   224→    });
   225→
   226→    created.push({
   227→      projectId: block.projectId,
   228→      projectName,
   229→      title,
   230→      members: block.items.length,
   231→    });
   232→  }
   233→  return created;
   234→}

   239→
   240→function synthesizeRollingSummary(
   241→  project: BrainProject,
   242→  items: BrainItem[],
   243→  decisions: BrainDecision[],
   244→  gaps: HealthGap[],
   245→  lastSessionTitle: string | null,
   246→  resolvedFixes: string[] = [],
   247→  activeFiles: string[] = []
   248→): string {
   249→  const typeCounts: Record<string, number> = {};
   250→  for (const it of items) typeCounts[it.itemType] = (typeCounts[it.itemType] ?? 0) + 1;
   251→  const typeStr = Object.entries(typeCounts)
   252→    .sort((a, b) => b[1] - a[1])
   253→    .map(([t, n]) => `${n} ${t}${n > 1 ? "s" : ""}`)
   254→    .join(", ");
   255→  const keywords = topKeywords(
   256→    items.map((i) => `${i.title} ${i.rawContent}`).join(" "),
   257→    8
   258→  );
   259→  const sessionLine = lastSessionTitle ? `\n- Latest work session: ${lastSessionTitle}` : "";
   260→
   261→  const lines: string[] = [];
   262→  lines.push(
   263→    `Compressed memory for **${project.name}** (${items.length} captures: ${typeStr || "none"}; auto-synthesized by the Memory Compression Daemon).`
   264→  );
   265→  lines.push(``);
   266→  lines.push(`## Current Focus`);
   267→  lines.push(
   268→    keywords.length > 0
   269→      ? `Active themes across recent captures: ${keywords.join(", ")}.`
   270→      : `No recent capture activity to derive themes from.`
   271→  );
   272→  lines.push(``);
   273→  lines.push(`## Recent Activity (last ${SUMMARY_WINDOW_DAYS} days)`);
   274→  lines.push(`- ${items.length} captures on record.${sessionLine}`);
   275→  lines.push(``);
   276→  lines.push(`## Active Decisions (do not violate)`);
   277→  if (decisions.length > 0) {
   278→    // v2.2: de-duplicate by normalized decision text. Repeated captures of the same
   279→    // choice log multiple Decision rows, and a degenerate 5x-same-decision list
   280→    // floods the Tier-1 char budget, crowding out fixes/gaps (and repeating itself
   281→    // verbatim in every injected prompt).
   282→    const seenDecisions = new Set<string>();
   283→    const uniqueDecisions: BrainDecision[] = [];
   284→    for (const d of decisions) {
   285→      const key = `${d.decision.trim().toLowerCase()}|${d.title.trim().toLowerCase()}`;
   286→      if (seenDecisions.has(key)) continue;
   287→      seenDecisions.add(key);
   288→      uniqueDecisions.push(d);
   289→      if (uniqueDecisions.length >= 5) break;
   290→    }
   291→    for (const d of uniqueDecisions) {
   292→      lines.push(`- **${d.title.slice(0, 60)}**: ${d.decision.slice(0, 80)} (Reason: ${d.reasoning.slice(0, 60)})`);
   293→    }
   294→  } else {
   295→    lines.push(`- None logged yet — capture choices with 'Decision: ... Reason: ...'`);
   296→  }
   297→  lines.push(``);
   298→  lines.push(`## Resolved Fixes`);
   299→  // v2.2: dedupe identical fix lines for the same reason as decisions above.
   300→  const uniqueFixes = [...new Set(resolvedFixes.map((f) => f.trim()).filter(Boolean))];
   301→  if (uniqueFixes.length > 0) {
   302→    for (const fx of uniqueFixes.slice(0, 4)) lines.push(`- ${fx.slice(0, 130)}`);
   303→  } else {
   304→    lines.push(`- No automated bug fixes logged yet.`);
   305→  }
   306→  if (activeFiles.length > 0) {
   307→    lines.push(``);
   308→    lines.push(`## Key Files on Record`);
   309→    lines.push(`- ${activeFiles.slice(0, 10).join(", ")}`);
   310→  }
   311→  if (gaps.length > 0) {
   312→    lines.push(``);
   313→    lines.push(`## Knowledge Gaps`);
   314→    for (const g of gaps) lines.push(`- [${g.severity.toUpperCase()}] ${g.gap} — ${g.detail}`);
   315→  }
   316→  return lines.join("\n").slice(0, MAX_SUMMARY_CHARS);
   317→}

   319→async function updateProjectSummaries(
   320→  trigger: CompressionReport["trigger"]
   321→): Promise<{ updated: string[]; gapReport: Record<string, HealthGap[]> }> {
   322→  const projects = await db.project.findMany({ orderBy: { updatedAt: "desc" } });
   323→  const updated: string[] = [];
   324→  const gapReport: Record<string, HealthGap[]> = {};
   325→
   326→  for (const prow of projects) {
   327→    const [itemRows, decisionRows] = await Promise.all([
   328→      db.item.findMany({
   329→        where: {
   330→          projectId: prow.id,
   331→          capturedAt: {
   332→            gte: new Date(Date.now() - SUMMARY_WINDOW_DAYS * 864e5),
   333→          },
   334→        },
   335→        orderBy: { capturedAt: "desc" },
   336→        take: 100,
   337→      }),
   338→      db.decision.findMany({
   339→        where: { projectId: prow.id },
   340→        orderBy: { decidedAt: "desc" },
   341→        take: 10,
   342→      }),
   343→    ]);
   344→
   345→    // Nothing to derive from — keep existing summary untouched
   346→    if (itemRows.length === 0) continue;
   347→
   348→    // Proper parsing via toItem keeps metadata a real object (fixes detection)
   349→    const items: BrainItem[] = itemRows.map((r) =>
   350→      toItem({ ...r, metadata: r.metadata })
   351→    );
   352→
   353→    const decisions: BrainDecision[] = decisionRows.map(toDecision);
   354→
   355→    const project: BrainProject = {
   356→      id: prow.id,
   357→      name: prow.name,
   358→      slug: prow.slug,
   359→      rootPath: prow.rootPath,
   360→      keywords: JSON.parse(prow.keywords || "[]") as string[],
   361→      gitRemotes: JSON.parse(prow.gitRemotes || "[]") as string[],
   362→      rollingSummary: prow.rollingSummary,
   363→      createdAt: prow.createdAt.toISOString(),
   364→      updatedAt: prow.updatedAt.toISOString(),
   365→    };
   366→
   367→    const edges = await getProjectGraph(prow.id, 50);
   368→    const { resolvedFixes, activeFiles } = extractFixesAndFiles(items, edges);
   369→    const { score, gaps } = computeHealthAndGaps(project, items, decisions, resolvedFixes);
   370→    gapReport[project.name] = gaps;
   371→
   372→    const lastSessionRow = await db.item.findFirst({
   373→      where: { projectId: prow.id, itemType: "session" },
   374→      orderBy: { capturedAt: "desc" },
   375→      select: { title: true },
   376→    });
   377→
   378→    const synthesized = synthesizeRollingSummary(
   379→      project,
   380→      items,
   381→      decisions,
   382→      gaps,
   383→      lastSessionRow?.title ?? null,
   384→      resolvedFixes,
   385→      activeFiles
   386→    );
   387→
   388→    // Only write when content actually changed (no churn on repeated runs)
   389→    if (synthesized !== prow.rollingSummary) {
   390→      await db.project.update({
   391→        where: { id: prow.id },
   392→        data: {
   393→          rollingSummary: synthesized,
   394→          healthScore: score,
   395→          healthGaps: JSON.stringify(gaps),
   396→          lastCompressedAt: new Date(),
   397→          updatedAt: new Date(),
   398→        },
   399→      });
   400→      updated.push(project.name);
   401→    } else {
   402→      // Persist health/lastCompressed even when summary is unchanged
   403→      await db.project.update({
   404→        where: { id: prow.id },
   405→        data: {
   406→          healthScore: score,
   407→          healthGaps: JSON.stringify(gaps),
   408→          lastCompressedAt: new Date(),
   409→        },
   410→      });
   411→    }
   412→  }
   413→
   414→  void trigger;
   415→  return { updated, gapReport };
   416→}

   422→export async function runCompressionCycle(
   423→  trigger: CompressionReport["trigger"] = "manual"
   424→): Promise<CompressionReport> {
   425→  if (inflight) return inflight; // never run two cycles concurrently
   426→  inflight = (async () => {
   427→    const startedAt = Date.now();
   428→    const backfill = await ensureVectorIndex();
   429→    const sessionDetails = await rollupSessions();
   430→    const { updated, gapReport } = await updateProjectSummaries(trigger);
   431→    const report: CompressionReport = {
   432→      ranAt: new Date().toISOString(),
   433→      durationMs: Date.now() - startedAt,
   434→      trigger,
   435→      sessionsCreated: sessionDetails.length,
   436→      sessionDetails,
   437→      summariesUpdated: updated,
   438→      gapReport,
   439→      vectorsBackfilled: backfill.backfilled,
   440→    };
   441→    lastRunAt = report.ranAt;
   442→    lastReport = report;
   443→    return report;
   444→  })();
   445→
   446→  try {
   447→    return await inflight;
   448→  } finally {
   449→    inflight = null;
   450→  }
   451→}

   453→// ==========================================
   454→// 3. PER-PROJECT COMPRESSION (v2, port of memory_compressor.compress_project_memory)
   455→// Kills token exhaustion: compresses notes, screenshots, decisions and bug fixes into a
   456→// dense summary (< 450 tokens), persists it as Tier-1 rolling memory, and writes
   457→// _project_summary.md + _work_replay.md into the real vault on disk.
   458→// ==========================================
   459→
   460→async function writeWorkSessionReplay(
   461→  projectSlug: string,
   462→  projectName: string,
   463→  items: BrainItem[]
   464→): Promise<string> {
   465→  const relPath = `ProjectBrainVault/Projects/${projectSlug}/_work_replay.md`;
   466→
   467→  // v2.1 (port of memory_compressor._write_work_session_replay): one regenerated file
   468→  // per compression — day blocks carry "N unique captures / M occurrences" headers,
   469→  // activity is grouped into 45-minute work sessions, and exact repeats fold into ×N.
   470→  const grouped = new Map<string, BrainItem[]>();
   471→  for (const it of items) {
   472→    const day = (it.capturedAt || "").slice(0, 10) || "Unknown-Date";
   473→    const arr = grouped.get(day) ?? [];
   474→    arr.push(it);
   475→    grouped.set(day, arr);
   476→  }
   477→
   478→  const occurrencesOf = (it: BrainItem): number =>
   479→    Math.max(1, Number(it.metadata?.occurrence_count ?? 1));
   480→
   481→  const lines: string[] = [
   482→    `# ${projectName} — Work Replay`,
   483→    "",
   484→    "> Regenerated from captures; repeated exact captures are folded into occurrence counts.",
   485→    "",
   486→  ];
   487→  const days = [...grouped.keys()].sort((a, b) => b.localeCompare(a));
   488→  for (const day of days) {
   489→    const dayItems = grouped.get(day)!;
   490→    const totalOccurrences = dayItems.reduce((acc, it) => acc + occurrencesOf(it), 0);
   491→    lines.push(
   492→      `## ${day} · ${dayItems.length} unique captures / ${totalOccurrences} occurrences`
   493→    );
   494→
   495→    // Chronological split into work sessions whenever the capture gap exceeds 45 minutes
   496→    const chrono = [...dayItems].sort((a, b) =>
   497→      (a.capturedAt || "").localeCompare(b.capturedAt || "")
   498→    );
   499→    const sessions: BrainItem[][] = [];
   500→    let current: BrainItem[] = [];
   501→    let priorMs: number | null = null;
   502→    for (const it of chrono) {
   503→      const stampMs = Date.parse(it.capturedAt || "");
   504→      if (
   505→        current.length > 0 &&
   506→        Number.isFinite(stampMs) &&
   507→        priorMs !== null &&
   508→        stampMs - priorMs > REPLAY_SESSION_GAP_MS
   509→      ) {
   510→        sessions.push(current);
   511→        current = [];
   512→      }
   513→      current.push(it);
   514→      if (Number.isFinite(stampMs)) priorMs = stampMs;
   515→    }
   516→    if (current.length > 0) sessions.push(current);
   517→
   518→    sessions.forEach((session, idx) => {
   519→      const first = (session[0].capturedAt || "").slice(11, 16);
   520→      const last = (session[session.length - 1].capturedAt || "").slice(11, 16);
   521→      lines.push("", `### Session ${idx + 1} · ${first}–${last}`);
   522→      for (const it of session) {
   523→        const count = occurrencesOf(it);
   524→        const suffix = count > 1 ? ` ×${count}` : "";
   525→        lines.push(
   526→          `- **[${it.itemType.toUpperCase()}] ${it.title}**${suffix} — ${it.structuredSummary || ""}`
   527→        );
   528→      }
   529→    });
   530→    lines.push("");
   531→  }
   532→
   533→  await writeVaultFile(relPath, lines.join("\n"));
   534→  return relPath;
   535→}

   537→export async function compressProjectMemory(
   538→  projectId: string,
   539→  opts: { skipDiskWrites?: boolean } = {}
   540→): Promise<ProjectCompressionReport | { error: string }> {
   541→  const prow = await db.project.findUnique({ where: { id: projectId } });
   542→  if (!prow) return { error: `Project '${projectId}' not found` };
   543→
   544→  const [itemRows, decisionRows] = await Promise.all([
   545→    db.item.findMany({
   546→      where: { projectId },
   547→      orderBy: { capturedAt: "desc" },
   548→      take: 100,
   549→    }),
   550→    db.decision.findMany({
   551→      where: { projectId },
   552→      orderBy: { decidedAt: "desc" },
   553→      take: 25,
   554→    }),
   555→  ]);
   556→  const edges = await getProjectGraph(projectId, 100);
   557→
   558→  const items: BrainItem[] = itemRows.map((r) => toItem(r));
   559→  const decisions: BrainDecision[] = decisionRows.map(toDecision);
   560→  const project: BrainProject = {
   561→    id: prow.id,
   562→    name: prow.name,
   563→    slug: prow.slug,
   564→    rootPath: prow.rootPath,
   565→    keywords: JSON.parse(prow.keywords || "[]") as string[],
   566→    gitRemotes: JSON.parse(prow.gitRemotes || "[]") as string[],
   567→    rollingSummary: prow.rollingSummary,
   568→    createdAt: prow.createdAt.toISOString(),
   569→    updatedAt: prow.updatedAt.toISOString(),
   570→  };
   571→
   572→  const notesCount = items.filter((i) => i.itemType === "note").length;
   573→  const shotsCount = items.filter((i) => i.itemType === "screenshot").length;
   574→  const termsCount = items.filter((i) => i.itemType === "terminal").length;
   575→  // v2.1: unique captures represent MORE observed occurrences after duplicate folding
   576→  const occurrencesRepresented = items.reduce(
   577→    (acc, it) => acc + Math.max(1, Number(it.metadata?.occurrence_count ?? 1)),
   578→    0
   579→  );
   580→
   581→  const { resolvedFixes, activeFiles } = extractFixesAndFiles(items, edges);
   582→  const { score, gaps } = computeHealthAndGaps(project, items, decisions, resolvedFixes);
   583→
   584→  const lastSessionRow = await db.item.findFirst({
   585→    where: { projectId, itemType: "session" },
   586→    orderBy: { capturedAt: "desc" },
   587→    select: { title: true },
   588→  });
   589→
   590→  // Deterministic compressed digest (works offline, zero API keys)
   591→  let finalSummary = synthesizeRollingSummary(
   592→    project,
   593→    items,
   594→    decisions,
   595→    gaps,
   596→    lastSessionRow?.title ?? null,
   597→    resolvedFixes,
   598→    activeFiles
   599→  );
   600→
   601→  // Optional live-LLM compression pass (Ollama / OpenAI) — falls back seamlessly
   602→  let llmDigestUsed = false;
   603→  const llmDigest = await llmBridge.completeText(
   604→    "Compress the following project state into a dense, factual paragraph (< 120 words) " +
   605→      "for an AI coding assistant. Preserve exact filenames, decisions, and package fixes.",
   606→    finalSummary,
   607→    200
   608→  );
   609→  if (llmDigest) {
   610→    finalSummary = llmDigest.slice(0, MAX_SUMMARY_CHARS);
   611→    llmDigestUsed = true;
   612→  }
   613→
   614→  const now = new Date();
   615→  await db.project.update({
   616→    where: { id: projectId },
   617→    data: {
   618→      rollingSummary: finalSummary,
   619→      healthScore: score,
   620→      healthGaps: JSON.stringify(gaps),
   621→      lastCompressedAt: now,
   622→      updatedAt: now,
   623→    },
   624→  });
   625→
   626→  // v2: write the enriched _project_summary.md + _work_replay.md to the real vault
   627→  const summaryRelPath = `ProjectBrainVault/Projects/${project.slug}/_project_summary.md`;
   628→  if (!opts.skipDiskWrites) {
   629→    const decisionsMd =
   630→      decisions
   631→        .map((d) => `- **${d.title}**: ${d.decision} *(Reason: ${d.reasoning})*`)
   632→        .join("\n") || "- None recorded yet.";
   633→    const fixesMd = resolvedFixes.map((fx) => `- ${fx}`).join("\n") || "- None recorded yet.";
   634→    const gapsMd = gaps.map((g) => `- ⚠️ ${g.gap} — ${g.detail}`).join("\n") || "- ✅ No knowledge gaps detected!";
   635→    const keywords = topKeywords(items.map((i) => `${i.title} ${i.rawContent}`).join(" "), 8);
   636→
   637→    await writeVaultFile(
   638→      summaryRelPath,
   639→      `# ${project.name} — Compressed Project Brain\n\n` +
   640→        `- **Health Score:** ${score}/100 (${healthStatus(score)})\n` +
   641→        `- **Unique Captures Indexed:** ${items.length} (${occurrencesRepresented} observed occurrences after duplicate folding)\n` +
   642→        `- **Composition:** ${notesCount} notes, ${shotsCount} screenshots, ${termsCount} terminal logs\n` +
   643→        `- **Decisions Logged:** ${decisions.length}\n` +
   644→        `- **Last Compressed:** ${now.toISOString()}\n\n` +
   645→        `## 1. Compressed Rolling Memory (Tier-1 Prompt Payload)\n${finalSummary}\n\n` +
   646→        `## 2. Key Architectural Decisions\n${decisionsMd}\n\n` +
   647→        `## 3. Known Bug Fixes & Actionable Remedies\n${fixesMd}\n\n` +
   648→        `## 4. Knowledge Gaps & Recommendations\n${gapsMd}\n` +
   649→        (activeFiles.length > 0 ? `\n## 5. Key Files\n- ${activeFiles.slice(0, 10).join(", ")}\n` : "") +
   650→        (keywords.length > 0 ? `\n_Topics: ${keywords.join(", ")}_\n` : "")
   651→    );
   652→  }
   653→  const replayRelPath = opts.skipDiskWrites
   654→    ? `ProjectBrainVault/Projects/${project.slug}/_work_replay.md`
   655→    : await writeWorkSessionReplay(project.slug, project.name, items);
   656→
   657→  return {
   658→    project_id: project.id,
   659→    project_name: project.name,
   660→    compressed_rolling_summary: finalSummary,
   661→    estimated_summary_tokens: Math.max(1, Math.floor(finalSummary.length / 4)),
   662→    health: {
   663→      score,
   664→      status: healthStatus(score),
   665→      gaps: gaps.map((g) => g.gap),
   666→    },
   667→    summary_file: summaryRelPath,
   668→    work_replay_file: replayRelPath,
   669→    items_compressed: items.length,
   670→    occurrences_represented: occurrencesRepresented,
   671→    llm_digest_used: llmDigestUsed,
   672→  };
   673→}

   675→// ==========================================
   676→// 4. CROSS-PROJECT LEARNING & BUG REUSE DETECTOR (v2)
   677→// Searches across ALL OTHER projects to check if a bug was already solved elsewhere.
   678→// ==========================================
   679→
   680→export async function findCrossProjectSolutions(
   681→  errorQuery: string,
   682→  excludeProjectId: string | null = null
   683→): Promise<CrossProjectSolution[]> {
   684→  const hits = await searchVault(errorQuery, null, 15);
   685→  const reusable: CrossProjectSolution[] = [];
   686→
   687→  for (const h of hits) {
   688→    if (excludeProjectId && h.projectId === excludeProjectId) continue;
   689→    let sourceProject = "Inbox";
   690→    if (h.projectId) {
   691→      const proj = await db.project.findUnique({ where: { id: h.projectId }, select: { name: true } });
   692→      sourceProject = proj?.name ?? "Inbox";
   693→    }
   694→    reusable.push({
   695→      item_id: h.id,
   696→      source_project: sourceProject,
   697→      title: h.title,
   698→      summary: h.structuredSummary,
   699→      suggested_fix: typeof h.metadata?.fix === "string" ? h.metadata.fix : "",
   700→      captured_at: h.capturedAt,
   701→    });
   702→  }
   703→  // v2.2: stable partition — items that actually carry an extracted fix ARE the
   704→  // reusable solutions, so they rank above fixless mentions. Order WITHIN each
   705→  // group stays exactly as searchVault ranked it (FTS relevance): re-sorting by
   706→  // recency here would let generic-advice captures ("Inspect stack trace for …")
   707→  // outrank a strongly-matching real fix from another project.
   708→  const withFix = reusable.filter((r) => r.suggested_fix);
   709→  const withoutFix = reusable.filter((r) => !r.suggested_fix);
   710→  return [...withFix, ...withoutFix];
   711→}
