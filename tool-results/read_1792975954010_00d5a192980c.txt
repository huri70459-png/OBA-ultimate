   110→/**
   111→ * Deterministic target positions for the guided layouts. Hierarchy: BFS layering
   112→ * from root candidates (no incoming edges, else the highest out-degree node),
   113→ * layered top-to-bottom. Radial: BFS rings around the highest-degree node.
   114→ * Pure function — safe for the render path (no randomness).
   115→ */
   116→export function computeLayoutTargets(
   117→  nodes: GraphViewerNode[],
   118→  links: GraphViewerLink[],
   119→  layout: GraphLayout,
   120→): Map<string, { x: number; y: number }> | null {
   121→  if (layout === "force" || nodes.length === 0) return null;
   122→  const ids = nodes.map((n) => n.id);
   123→  const outAdj = new Map<string, string[]>();
   124→  const inDeg = new Map<string, number>();
   125→  for (const id of ids) {
   126→    outAdj.set(id, []);
   127→    inDeg.set(id, 0);
   128→  }
   129→  for (const l of links) {
   130→    if (!outAdj.has(l.source) || !outAdj.has(l.target) || l.source === l.target) continue;
   131→    outAdj.get(l.source)!.push(l.target);
   132→    inDeg.set(l.target, (inDeg.get(l.target) ?? 0) + 1);
   133→  }
   134→  const targets = new Map<string, { x: number; y: number }>();
   135→
   136→  if (layout === "hierarchy") {
   137→    let roots = ids.filter((id) => (inDeg.get(id) ?? 0) === 0);
   138→    if (roots.length === 0) {
   139→      // Cyclic graph: seed from the most connected node.
   140→      roots = [ids.reduce((a, b) => (outAdj.get(a)!.length >= outAdj.get(b)!.length ? a : b))];
   141→    }
   142→    const layerOf = new Map<string, number>();
   143→    const queue = roots.map((id) => {
   144→      layerOf.set(id, 0);
   145→      return id;
   146→    });
   147→    for (let qi = 0; qi < queue.length; qi++) {
   148→      const cur = queue[qi];
   149→      const layer = layerOf.get(cur) ?? 0;
   150→      for (const next of outAdj.get(cur) ?? []) {
   151→        if (!layerOf.has(next)) {
   152→          layerOf.set(next, layer + 1);
   153→          queue.push(next);
   154→        }
   155→      }
   156→    }
   157→    const maxLayer = Math.max(0, ...[...layerOf.values()]);
   158→    const buckets: string[][] = Array.from({ length: maxLayer + 1 }, () => []);
   159→    for (const id of ids) buckets[layerOf.get(id) ?? maxLayer].push(id);
   160→    buckets.forEach((bucket, layer) => {
   161→      bucket.forEach((id, i) => {
   162→        targets.set(id, {
   163→          x: (i - (bucket.length - 1) / 2) * 120,
   164→          y: layer * 110,
   165→        });
   166→      });
   167→    });
   168→    return targets;
   169→  }
   170→
   171→  // radial — BFS rings around the most connected node
   172→  const totalDeg = (id: string) => (outAdj.get(id)?.length ?? 0) + (inDeg.get(id) ?? 0);
   173→  const center = ids.reduce((a, b) => (totalDeg(a) >= totalDeg(b) ? a : b));
   174→  const ringOf = new Map<string, number>([[center, 0]]);
   175→  const queue = [center];
   176→  while (queue.length > 0) {
   177→    const cur = queue.shift()!;
   178→    const ring = ringOf.get(cur)!;
   179→    for (const next of outAdj.get(cur) ?? []) {
   180→      if (!ringOf.has(next)) {
   181→        ringOf.set(next, ring + 1);
   182→        queue.push(next);
   183→      }
   184→    }
   185→    for (const id of inDeg.keys()) {
   186→      // also walk incoming edges so directed-only backlinks land on a ring
   187→      if (id === cur || ringOf.has(id)) continue;
   188→      if ((outAdj.get(id) ?? []).includes(cur)) {
   189→        ringOf.set(id, ring + 1);
   190→        queue.push(id);
   191→      }
   192→    }
   193→  }
   194→  const maxRing = Math.max(0, ...[...ringOf.values()]);
   195→  const rings: string[][] = Array.from({ length: maxRing + 1 }, () => []);
   196→  for (const id of ids) rings[ringOf.get(id) ?? maxRing].push(id);
   197→  rings.forEach((ring, d) => {
   198→    const radius = d === 0 ? 0 : 90 + d * 95;
   199→    ring.forEach((id, i) => {
   200→      if (d === 0) {
   201→        targets.set(id, { x: 0, y: 0 });
   202→        return;
   203→      }
   204→      const angle = (i / ring.length) * Math.PI * 2 + d * 0.55;
   205→      targets.set(id, { x: Math.cos(angle) * radius, y: Math.sin(angle) * radius });
   206→    });
   207→  });
   208→  return targets;
   209→}
   210→
   211→/**
   212→ * One animation frame of the layout: pairwise repulsion (+soft collision), link
   213→ * springs, then EITHER guided-target pull (hierarchy/radial) or gravity toward
   214→ * the origin (force), followed by damped integration. Reads node data fresh from
   215→ * propsRef every frame so data edits apply mid-animation.
   216→ */
   217→function runSimulation(ctx: SimContext) {
   218→  const { nodes, links } = ctx.propsRef.current;
   219→  const sims = ctx.sims;
   220→  const list: Array<{ id: string; s: Sim }> = [];
   221→  for (const n of nodes) {
   222→    const s = sims.get(n.id);
   223→    if (s) list.push({ id: n.id, s });
   224→  }
   225→
   226→  // node repulsion (pairwise)
   227→  for (let i = 0; i < list.length; i++) {
   228→    const a = list[i].s;
   229→    for (let j = i + 1; j < list.length; j++) {
   230→      const b = list[j].s;
   231→      const dx = a.x - b.x;
   232→      const dy = a.y - b.y;
   233→      const d2 = dx * dx + dy * dy + 0.01;
   234→      const d = Math.sqrt(d2);
   235→      let f = REPULSION / d2;
   236→      if (d < MIN_GAP) f += (MIN_GAP - d) * 0.06; // soft collision
   237→      const fx = (dx / d) * f;
   238→      const fy = (dy / d) * f;
   239→      a.vx += fx;
   240→      a.vy += fy;
   241→      b.vx -= fx;
   242→      b.vy -= fy;
   243→    }
   244→  }
   245→
   246→  // link springs
   247→  for (const l of links) {
   248→    const a = sims.get(l.source);
   249→    const b = sims.get(l.target);
   250→    if (!a || !b) continue;
   251→    const dx = b.x - a.x;
   252→    const dy = b.y - a.y;
   253→    const d = Math.hypot(dx, dy) || 1;
   254→    const f = (d - LINK_REST) * LINK_STRENGTH;
   255→    const fx = (dx / d) * f;
   256→    const fy = (dy / d) * f;
   257→    a.vx += fx;
   258→    a.vy += fy;
   259→    b.vx -= fx;
   260→    b.vy -= fy;
   261→  }
   262→
   263→  // gravity or guided-target pull + integrate
   264→  const targets = ctx.targets.current;
   265→  for (const { id, s } of list) {
   266→    const target = targets?.get(id);
   267→    if (target) {
   268→      s.vx += (target.x - s.x) * 0.055;
   269→      s.vy += (target.y - s.y) * 0.055;
   270→    } else {
   271→      s.vx += -s.x * GRAVITY;
   272→      s.vy += -s.y * GRAVITY;
   273→    }
   274→    if (s.fx !== null && s.fy !== null) {
   275→      s.x = s.fx;
   276→      s.y = s.fy;
   277→      s.vx = 0;
   278→      s.vy = 0;
   279→    } else {
   280→      s.vx *= DAMPING;
   281→      s.vy *= DAMPING;
   282→      s.x += s.vx;
   283→      s.y += s.vy;
   284→    }
   285→  }
   286→
   287→  // Publish a fresh render snapshot (keeps the React render path ref-free)
   288→  const snapshot: Record<string, { x: number; y: number }> = {};
   289→  for (const { id, s } of list) {
   290→    snapshot[id] = { x: s.x, y: s.y };
   291→  }
   292→  ctx.setPositions(snapshot);
   293→  ctx.alphaRef.current *= ALPHA_DECAY;
   294→  ctx.rafRef.current =
   295→    ctx.alphaRef.current > MIN_ALPHA ? requestAnimationFrame(() => runSimulation(ctx)) : null;
   296→}

   298→export const GraphViewer = ({
   299→  nodes,
   300→  links,
   301→  onNodeClick,
   302→  selectedId = null,
   303→  height = 460,
   304→  className = "",
   305→  ariaLabel = "Interactive knowledge graph",
   306→  loading = false,
   307→}: GraphViewerProps) => {
   308→  const { t } = useI18n();
   309→  const svgRef = useRef<SVGSVGElement | null>(null);
   310→
   311→  // ---- layout mode (force | hierarchy | radial) ----
   312→  const [layout, setLayout] = useState<GraphLayout>("force");
   313→  const targetsRef = useRef<Map<string, { x: number; y: number }> | null>(null);
   314→
   315→  // ---- simulation state (refs: mutated by the rAF loop; painted via state) ----
   316→  const simRef = useRef<Map<string, Sim>>(new Map());
   317→  const alphaRef = useRef(0);
   318→  const rafRef = useRef<number | null>(null);
   319→  const propsRef = useRef({ nodes, links });
   320→  // Sync latest graph data outside of render (react-hooks/refs-compliant)
   321→  useEffect(() => {
   322→    propsRef.current = { nodes, links };
   323→  }, [nodes, links]);
   324→
   325→  // Render-side mirror of the live simulation — the loop publishes a fresh
   326→  // snapshot each frame, keeping the render path ref-free (react-hooks/refs).
   327→  const [positions, setPositions] = useState<Record<string, { x: number; y: number }>>({});
   328→
   329→  const [transform, setTransform] = useState<Transform>({ x: 460, y: 230, k: 1 });
   330→  const [hoverId, setHoverId] = useState<string | null>(null);
   331→  const dragRef = useRef<{ id: string; dx: number; dy: number; moved: boolean } | null>(null);
   332→  const panRef = useRef<{ px: number; py: number; tx: number; ty: number; moved: boolean } | null>(null);
   333→
   334→  // ---- degree + radius (memoized) ----
   335→  const degree = useMemo(() => {
   336→    const d = new Map<string, number>();
   337→    for (const l of links) {
   338→      d.set(l.source, (d.get(l.source) ?? 0) + 1);
   339→      d.set(l.target, (d.get(l.target) ?? 0) + 1);
   340→    }
   341→    return d;
   342→  }, [links]);
   343→
   344→  const radiusOf = (id: string) => {
   345→    const deg = degree.get(id) ?? 0;
   346→    return 9 + Math.min(9, deg * 1.7);
   347→  };
   348→
   349→  // ---- adjacency for hover highlighting ----
   350→  const neighbors = useMemo(() => {
   351→    const m = new Map<string, Set<string>>();
   352→    const add = (a: string, b: string) => {
   353→      if (!m.has(a)) m.set(a, new Set());
   354→      m.get(a)!.add(b);
   355→    };
   356→    for (const l of links) {
   357→      add(l.source, l.target);
   358→      add(l.target, l.source);
   359→    }
   360→    return m;
   361→  }, [links]);
   362→
   363→  // ---- deterministic initial layout (golden-angle spiral; pure render-safe) ----
   364→  const ensureSims = useCallback(() => {
   365→    const sims = simRef.current;
   366→    propsRef.current.nodes.forEach((n, i) => {
   367→      if (sims.has(n.id)) return;
   368→      const angle = i * 2.399963; // golden angle
   369→      const r = 60 + Math.sqrt(i + 1) * 46;
   370→      sims.set(n.id, {
   371→        x: Math.cos(angle) * r,
   372→        y: Math.sin(angle) * r,
   373→        vx: 0,
   374→        vy: 0,
   375→        fx: null,
   376→        fy: null,
   377→      });
   378→    });
   379→    for (const id of [...sims.keys()]) {
   380→      if (!propsRef.current.nodes.some((n) => n.id === id)) sims.delete(id);
   381→    }
   382→  }, []);
   383→
   384→  // ---- force tick (module-scope loop; self-recursion lives outside React) ----
   385→  const tick = useCallback(() => {
   386→    runSimulation({ propsRef, sims: simRef.current, alphaRef, rafRef, setPositions, targets: targetsRef });
   387→  }, [setPositions]);
   388→
   389→  const reheat = useCallback(
   390→    (alpha: number) => {
   391→      alphaRef.current = Math.max(alphaRef.current, alpha);
   392→      if (rafRef.current === null) rafRef.current = requestAnimationFrame(tick);
   393→    },
   394→    [tick],
   395→  );
   396→
   397→  // (re)initialize on data or layout change
   398→  useEffect(() => {
   399→    targetsRef.current = computeLayoutTargets(nodes, links, layout);
   400→    ensureSims();
   401→    reheat(1);
   402→  }, [nodes, links, layout, ensureSims, reheat]);
   403→
   404→  // Stop the loop on unmount. rafRef MUST be reset to null: React StrictMode
   405→  // runs effect -> cleanup -> effect in dev, and a stale non-null id would make
   406→  // reheat() believe a frame is still pending, permanently killing the layout.
   407→  useEffect(() => {
   408→    return () => {
   409→      if (rafRef.current !== null) {
   410→        cancelAnimationFrame(rafRef.current);
   411→        rafRef.current = null;
   412→      }
   413→    };
   414→  }, []);
   415→
   416→  // ---- zoom (non-passive wheel listener for preventDefault) ----
   417→  const zoomAt = useCallback((clientX: number, clientY: number, factor: number) => {
   418→    const el = svgRef.current;
   419→    if (!el) return;
   420→    const rect = el.getBoundingClientRect();
   421→    const px = clientX - rect.left;
   422→    const py = clientY - rect.top;
   423→    setTransform((tr) => {
   424→      const k = clamp(tr.k * factor, 0.2, 3.2);
   425→      return { k, x: px - ((px - tr.x) / tr.k) * k, y: py - ((py - tr.y) / tr.k) * k };
   426→    });
   427→  }, []);
   428→
   429→  useEffect(() => {
   430→    const el = svgRef.current;
   431→    if (!el) return;
   432→    const onWheel = (e: WheelEvent) => {
   433→      e.preventDefault();
   434→      zoomAt(e.clientX, e.clientY, Math.exp(-e.deltaY * 0.0016));
   435→    };
   436→    el.addEventListener("wheel", onWheel, { passive: false });
   437→    return () => el.removeEventListener("wheel", onWheel);
   438→  }, [zoomAt]);
   439→
   440→  const fitView = useCallback(() => {
   441→    const el = svgRef.current;
   442→    if (!el) return;
   443→    const rect = el.getBoundingClientRect();
   444→    const sims = [...simRef.current.values()];
   445→    if (sims.length === 0 || rect.width === 0) return;
   446→    const minX = Math.min(...sims.map((s) => s.x));
   447→    const maxX = Math.max(...sims.map((s) => s.x));
   448→    const minY = Math.min(...sims.map((s) => s.y));
   449→    const maxY = Math.max(...sims.map((s) => s.y));
   450→    const bw = Math.max(maxX - minX, 80);
   451→    const bh = Math.max(maxY - minY, 80);
   452→    const k = clamp(Math.min(rect.width / bw, rect.height / bh) * 0.82, 0.25, 1.8);
   453→    setTransform({
   454→      k,
   455→      x: rect.width / 2 - ((minX + maxX) / 2) * k,
   456→      y: rect.height / 2 - ((minY + maxY) / 2) * k,
   457→    });
   458→  }, []);
   459→
   460→  const zoomBy = (factor: number) => {
   461→    const el = svgRef.current;
   462→    if (!el) return;
   463→    const rect = el.getBoundingClientRect();
   464→    zoomAt(rect.left + rect.width / 2, rect.top + rect.height / 2, factor);
   465→  };
   466→
   467→  /** Blueprint "Zoom & Pan Controls": slider maps 20..320 percent onto zoom k. */
   468→  const onSliderZoom = (values: number[]) => {
   469→    const el = svgRef.current;
   470→    if (!el) return;
   471→    const rect = el.getBoundingClientRect();
   472→    const desired = clamp(values[0] / 100, 0.2, 3.2);
   473→    zoomAt(rect.left + rect.width / 2, rect.top + rect.height / 2, desired / transform.k);
   474→  };
   475→
   476→  // ---- pointer interactions ----
   477→  const toGraph = (clientX: number, clientY: number) => {
   478→    const el = svgRef.current;
   479→    const rect = el!.getBoundingClientRect();
   480→    return {
   481→      gx: (clientX - rect.left - transform.x) / transform.k,
   482→      gy: (clientY - rect.top - transform.y) / transform.k,
   483→    };
   484→  };
   485→
   486→  const capturePointer = (e: React.PointerEvent, el: Element) => {
   487→    try {
   488→      el.setPointerCapture(e.pointerId);
   489→    } catch {
   490→      // Synthetic pointers (tests, some touch stacks) have no active pointer id —
   491→      // capture is an optimization here, not a requirement.
   492→    }
   493→  };
   494→
   495→  const onPointerDownNode = (e: React.PointerEvent, id: string) => {
   496→    e.stopPropagation();
   497→    capturePointer(e, e.currentTarget as Element);
   498→    const { gx, gy } = toGraph(e.clientX, e.clientY);
   499→    const s = simRef.current.get(id);
   500→    if (!s) return;
   501→    dragRef.current = { id, dx: gx - s.x, dy: gy - s.y, moved: false };
   502→    s.fx = gx - dragRef.current.dx;
   503→    s.fy = gy - dragRef.current.dy;
   504→    reheat(0.45);
   505→  };
   506→
   507→  const onPointerDownBg = (e: React.PointerEvent) => {
   508→    capturePointer(e, e.currentTarget as Element);
   509→    panRef.current = { px: e.clientX, py: e.clientY, tx: transform.x, ty: transform.y, moved: false };
   510→  };
   511→
   512→  const onPointerMove = (e: React.PointerEvent) => {
   513→    if (dragRef.current) {
   514→      const { gx, gy } = toGraph(e.clientX, e.clientY);
   515→      const s = simRef.current.get(dragRef.current.id);
   516→      if (s) {
   517→        s.fx = gx - dragRef.current.dx;
   518→        s.fy = gy - dragRef.current.dy;
   519→        dragRef.current.moved = true;
   520→        reheat(0.4);
   521→      }
   522→      return;
   523→    }
   524→    if (panRef.current) {
   525→      const dx = e.clientX - panRef.current.px;
   526→      const dy = e.clientY - panRef.current.py;
   527→      if (Math.abs(dx) + Math.abs(dy) > 3) panRef.current.moved = true;
   528→      setTransform((tr) => ({ ...tr, x: panRef.current!.tx + dx, y: panRef.current!.ty + dy }));
   529→    }
   530→  };
   531→
   532→  const onPointerUp = () => {
   533→    if (dragRef.current) {
   534→      const { id, moved } = dragRef.current;
   535→      const s = simRef.current.get(id);
   536→      if (s) {
   537→        s.fx = null;
   538→        s.fy = null;
   539→      }
   540→      if (!moved && onNodeClick) {
   541→        const node = propsRef.current.nodes.find((n) => n.id === id);
   542→        if (node) onNodeClick(node);
   543→      }
   544→      dragRef.current = null;
   545→    }
   546→    panRef.current = null;
   547→  };
   548→
   549→  // ---- rendering ----
   550→  const isDimmed = (id: string) =>
   551→    hoverId !== null && id !== hoverId && !(neighbors.get(hoverId)?.has(id) ?? false);
   552→
   553→  return (
   554→    <div className={`relative overflow-hidden rounded-xl border border-border bg-[var(--graph-canvas)] ${className}`}>
   555→      <svg
   556→        ref={svgRef}
   557→        role="application"
   558→        aria-label={ariaLabel}
   559→        style={{ height, touchAction: "none" }}
   560→        className="w-full cursor-grab active:cursor-grabbing"
   561→        onPointerDown={onPointerDownBg}
   562→        onPointerMove={onPointerMove}
   563→        onPointerUp={onPointerUp}
   564→        onPointerLeave={onPointerUp}
   565→      >
   566→        <defs>
   567→          <pattern id="graph-grid" width="26" height="26" patternUnits="userSpaceOnUse">
   568→            <circle cx="1" cy="1" r="1" fill="var(--graph-grid)" />
   569→          </pattern>
   570→          {RELATIONS.map((rel) => (
   571→            <marker
   572→              key={rel}
   573→              id={`gv-arrow-${rel}`}
   574→              viewBox="0 0 10 10"
   575→              refX="9"
   576→              refY="5"
   577→              markerWidth="6"
   578→              markerHeight="6"
   579→              orient="auto-start-reverse"
   580→            >
   581→              <path d="M 0 0 L 10 5 L 0 10 z" fill={relationColor(rel)} />
   582→            </marker>
   583→          ))}
   584→          <marker id="gv-arrow-default" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6" markerHeight="6" orient="auto-start-reverse">
   585→            <path d="M 0 0 L 10 5 L 0 10 z" fill="var(--graph-edge)" />
   586→          </marker>
   587→        </defs>
   588→        <rect width="100%" height="100%" fill="url(#graph-grid)" opacity={0.5} />
   589→
   590→        <g transform={`translate(${transform.x},${transform.y}) scale(${transform.k})`}>
   591→          {/* edges */}
   592→          {links.map((l, i) => {
   593→            const a = positions[l.source];
   594→            const b = positions[l.target];
   595→            if (!a || !b) return null;
   596→            const dx = b.x - a.x;
   597→            const dy = b.y - a.y;
   598→            const len = Math.hypot(dx, dy) || 1;
   599→            const ox = (-dy / len) * len * 0.1;
   600→            const oy = (dx / len) * len * 0.1;
   601→            const cx = (a.x + b.x) / 2 + ox;
   602→            const cy = (a.y + b.y) / 2 + oy;
   603→            // pull the end back so the arrowhead meets the node ring
   604→            const ex = cx - b.x;
   605→            const ey = cy - b.y;
   606→            const el = Math.hypot(ex, ey) || 1;
   607→            const r2 = radiusOf(l.target);
   608→            const endX = b.x + (ex / el) * (r2 + 5);
   609→            const endY = b.y + (ey / el) * (r2 + 5);
   610→            const rel = l.relation ?? "";
   611→            const highlighted = hoverId !== null && (l.source === hoverId || l.target === hoverId);
   612→            const relatedToSelection =
   613→              selectedId !== null && (l.source === selectedId || l.target === selectedId);
   614→            return (
   615→              <path
   616→                key={l.id ?? `${l.source}->${l.target}-${i}`}
   617→                d={`M ${a.x} ${a.y} Q ${cx} ${cy} ${endX} ${endY}`}
   618→                fill="none"
   619→                stroke={rel && RELATIONS.includes(rel) ? relationColor(rel) : "var(--graph-edge)"}
   620→                strokeWidth={highlighted || relatedToSelection ? 2 : 1.2}
   621→                strokeOpacity={highlighted || relatedToSelection ? 0.95 : 0.45}
   622→                markerEnd={rel && RELATIONS.includes(rel) ? `url(#gv-arrow-${rel})` : "url(#gv-arrow-default)"}
   623→              />
   624→            );
   625→          })}
   626→
   627→          {/* nodes */}
   628→          {nodes.map((n) => {
   629→            const p = positions[n.id];
   630→            if (!p) return null;
   631→            const r = radiusOf(n.id);
   632→            const kindColor = KIND_COLOR[n.kind ?? "concept"] ?? KIND_COLOR.concept;
   633→            const dim = isDimmed(n.id);
   634→            const selected = selectedId === n.id;
   635→            return (
   636→              <g
   637→                key={n.id}
   638→                opacity={dim ? 0.22 : 1}
   639→                onPointerDown={(e) => onPointerDownNode(e, n.id)}
   640→                onPointerEnter={() => !dragRef.current && !panRef.current && setHoverId(n.id)}
   641→                onPointerLeave={() => setHoverId((h) => (h === n.id ? null : h))}
   642→                style={{ cursor: "pointer" }}
   643→              >
   644→                {selected && (
   645→                  <circle cx={p.x} cy={p.y} r={r + 5} fill="none" stroke="var(--primary)" strokeWidth={1.4} strokeOpacity={0.9} />
   646→                )}
   647→                <circle
   648→                  cx={p.x}
   649→                  cy={p.y}
   650→                  r={r}
   651→                  fill="var(--graph-node-fill)"
   652→                  stroke={kindColor}
   653→                  strokeWidth={selected || hoverId === n.id ? 2.4 : 1.6}
   654→                />
   655→                <circle cx={p.x} cy={p.y} r={r - 3.5} fill={kindColor} fillOpacity={0.18} />
   656→                <text
   657→                  x={p.x}
   658→                  y={p.y + r + 12}
   659→                  textAnchor="middle"
   660→                  fontSize={10}
   661→                  fontFamily="var(--font-geist-mono), monospace"
   662→                  fill="var(--graph-node-text)"
   663→                  opacity={0.92}
   664→                  style={{ pointerEvents: "none", userSelect: "none" }}
   665→                >
   666→                  {truncate(n.label || n.id, LABEL_MAX)}
   667→                </text>
   668→              </g>
   669→            );
   670→          })}
   671→        </g>
   672→      </svg>
   673→
   674→      {/* controls */}
   675→      <div className="absolute right-2.5 top-2.5 flex flex-col gap-1">
   676→        <ControlButton label="Zoom in" onClick={() => zoomBy(1.25)}>
   677→          <ZoomIn className="h-3.5 w-3.5" />
   678→        </ControlButton>
   679→        <ControlButton label="Zoom out" onClick={() => zoomBy(0.8)}>
   680→          <ZoomOut className="h-3.5 w-3.5" />
   681→        </ControlButton>
   682→        <ControlButton label="Fit view" onClick={fitView}>
   683→          <Maximize className="h-3.5 w-3.5" />
   684→        </ControlButton>
   685→        <ControlButton label="Re-run layout" onClick={() => reheat(1)}>
   686→          <RefreshCw className="h-3.5 w-3.5" />
   687→        </ControlButton>
   688→      </div>
   689→
   690→      {/* layout selector + zoom slider (blueprint control set) */}
   691→      <div className="absolute bottom-2 right-2.5 flex items-center gap-2 rounded-lg border border-border bg-card/90 px-2 py-1.5 backdrop-blur">
   692→        <select
   693→          value={layout}
   694→          onChange={(e) => setLayout(e.target.value as GraphLayout)}
   695→          aria-label="Graph layout algorithm"
   696→          className="cursor-pointer rounded border border-border bg-background/70 px-1.5 py-0.5 text-[10px] text-foreground focus:outline-none"
   697→        >
   698→          <option value="force">{t("graph.layout.force")}</option>
   699→          <option value="hierarchy">{t("graph.layout.hierarchy")}</option>
   700→          <option value="radial">{t("graph.layout.radial")}</option>
   701→        </select>
   702→        <button
   703→          type="button"
   704→          onClick={() => zoomBy(0.8)}
   705→          aria-label="Zoom out"
   706→          className="text-muted-foreground transition-colors hover:text-foreground"
   707→        >
   708→          <ZoomOut className="h-3 w-3" />
   709→        </button>
   710→        <Slider
   711→          value={[Math.round(transform.k * 100)]}
   712→          min={20}
   713→          max={320}
   714→          step={5}
   715→          onValueChange={onSliderZoom}
   716→          aria-label="Zoom level"
   717→          className="w-20"
   718→        />
   719→        <button
   720→          type="button"
   721→          onClick={() => zoomBy(1.25)}
   722→          aria-label="Zoom in"
   723→          className="text-muted-foreground transition-colors hover:text-foreground"
   724→        >
   725→          <ZoomIn className="h-3 w-3" />
   726→        </button>
   727→        <span className="w-8 text-right text-[10px] tabular-nums text-muted-foreground">
   728→          {Math.round(transform.k * 100)}%
   729→        </span>
   730→      </div>
   731→
   732→      {loading && (
   733→        <div
   734→          className="absolute inset-0 z-10 flex items-center justify-center bg-card/80 backdrop-blur-sm"
   735→          role="status"
   736→          aria-label="Loading graph"
   737→        >
   738→          <Loader2 className="h-7 w-7 animate-spin text-primary" />
   739→        </div>
   740→      )}
   741→
   742→      <div className="pointer-events-none absolute bottom-2 left-3 text-[10px] text-muted-foreground/70">
   743→        drag nodes · scroll to zoom · drag canvas to pan · click a node to inspect
   744→      </div>
   745→    </div>
   746→  );
   747→};

   749→const ControlButton = ({
   750→  label,
   751→  onClick,
   752→  children,
   753→}: {
   754→  label: string;
   755→  onClick: () => void;
   756→  children: React.ReactNode;
   757→}) => (
   758→  <button
   759→    type="button"
   760→    onClick={onClick}
   761→    title={label}
   762→    aria-label={label}
   763→    className="flex h-7 w-7 items-center justify-center rounded-md border border-border bg-card/90 text-muted-foreground transition-colors hover:bg-accent hover:text-accent-foreground"
   764→  >
   765→    {children}
   766→  </button>
   767→);
   768→
   769→function relationColor(rel: string): string {
   770→  const map: Record<string, string> = {
   771→    FIXED_BY: "var(--rel-fixed-by)",
   772→    DECIDED_ON: "var(--rel-decided-on)",
   773→    MENTIONED_IN: "var(--rel-mentioned-in)",
   774→    RELATED_TO: "var(--rel-related-to)",
   775→  };
   776→  return map[rel] ?? "var(--graph-edge)";
   777→}
