   121→export async function writeVaultFile(relativePath: string, content: string): Promise<string | null> {
   122→  try {
   123→    const abs = resolveVaultPath(relativePath);
   124→    await mkdir(path.dirname(abs), { recursive: true });
   125→    await writeFile(abs, content, "utf-8");
   126→    return relativePath;
   127→  } catch {
   128→    return null; // never let disk failures break the engine
   129→  }

   187→function safeParseArray(json: string | null | undefined): string[] {
   188→  try {
   189→    const parsed = JSON.parse(json || "[]");
   190→    return Array.isArray(parsed) ? parsed.map(String) : [];

   192→    return [];
   193→  }

   209→export async function upsertProject(input: {
   210→  name: string;
   211→  rootPath?: string;
   212→  keywords?: string[];
   213→  gitRemotes?: string[];
   214→  rollingSummary?: string;
   215→}): Promise<BrainProject> {
   216→  const slug = slugify(input.name);
   217→  const keywords = input.keywords ?? [];
   218→  const gitRemotes = input.gitRemotes ?? [];
   219→
   220→  const existing = await db.project.findUnique({ where: { slug } });
   221→  let project: BrainProject;
   222→
   223→  if (existing) {
   224→    const updated = await db.project.update({
   225→      where: { id: existing.id },
   226→      data: {
   227→        name: input.name,
   228→        rootPath: input.rootPath ?? existing.rootPath,
   229→        keywords: JSON.stringify(keywords),
   230→        gitRemotes: JSON.stringify(gitRemotes),
   231→        rollingSummary: input.rollingSummary || existing.rollingSummary,
   232→        updatedAt: new Date(),
   233→      },
   234→    });
   235→    project = toProject(updated);
   236→  } else {
   237→    const created = await db.project.create({
   238→      data: {
   239→        id: `proj_${randomUUID().replace(/-/g, "").slice(0, 10)}`,
   240→        name: input.name,
   241→        slug,
   242→        rootPath: input.rootPath ?? "",
   243→        keywords: JSON.stringify(keywords),
   244→        gitRemotes: JSON.stringify(gitRemotes),
   245→        rollingSummary: input.rollingSummary ?? "",
   246→      },
   247→    });
   248→    project = toProject(created);
   249→  }
   250→
   251→  // v2: maintain the human-readable _project_summary.md on disk (sqlite_store.py parity:
   252→  // writes when a summary exists or the file is missing)
   253→  const effectiveSummary = project.rollingSummary;
   254→  const summaryRelPath = `ProjectBrainVault/Projects/${project.slug}/_project_summary.md`;
   255→  const summaryAbs = resolveVaultPath(summaryRelPath);
   256→  let summaryExists = true;
   257→  try {
   258→    await readFile(summaryAbs, "utf-8");
   259→  } catch {
   260→    summaryExists = false;
   261→  }
   262→  if (effectiveSummary || !summaryExists) {
   263→    const now = new Date().toISOString();
   264→    await writeVaultFile(
   265→      summaryRelPath,
   266→      `# ${project.name} — Project Brain Summary\n\n` +
   267→        `**Root Path:** \`${project.rootPath}\`\n` +
   268→        `**Keywords:** ${project.keywords.join(", ")}\n` +
   269→        `**Last Updated:** ${now}\n\n` +
   270→        `## Rolling Memory\n${effectiveSummary || "No summary compressed yet."}\n`
   271→    );
   272→  }
   273→
   274→  return project;
   275→}

   287→/** Resolves proj_xxx id, slug, or display name -> project id (port of mcp resolve_project_id). */
   288→export async function resolveProjectId(identifier: string): Promise<string | null> {
   289→  if (!identifier) return null;
   290→  const projects = await getAllProjects();
   291→  for (const p of projects) {
   292→    if (p.id === identifier || p.slug === identifier || p.name === identifier) return p.id;
   293→  }
   294→  const lower = identifier.toLowerCase();
   295→  for (const p of projects) {
   296→    if (p.name.toLowerCase() === lower || p.slug === lower) return p.id;
   297→  }
   298→  return null;
   299→}

   305→function vaultFilePath(project: BrainProject | null, itemType: string, title: string, now: Date): string {
   306→  const stamp = `${now.getUTCFullYear()}${String(now.getUTCMonth() + 1).padStart(2, "0")}${String(
   307→    now.getUTCDate()
   308→  ).padStart(2, "0")}_${String(now.getUTCHours()).padStart(2, "0")}${String(now.getUTCMinutes()).padStart(
   309→    2,
   310→    "0"
   311→  )}${String(now.getUTCSeconds()).padStart(2, "0")}`;
   312→  const fileSlug = slugify(title).slice(0, 50);
   313→  const filename = `${stamp}_${fileSlug}.md`;
   314→  if (project) {
   315→    const subfolder = ["screenshot", "clip", "terminal"].includes(itemType) ? "captures" : "notes";
   316→    return `ProjectBrainVault/Projects/${project.slug}/${subfolder}/${filename}`;
   317→  }
   318→  return `ProjectBrainVault/Inbox/${filename}`;
   319→}

   321→export async function saveItem(input: {
   322→  title: string;
   323→  rawContent: string;
   324→  itemType?: string;
   325→  projectId?: string | null;
   326→  structuredSummary?: string;
   327→  visualDescription?: string;
   328→  assetPath?: string;
   329→  sourceApp?: string;
   330→  sourceUrl?: string;
   331→  metadata?: Record<string, unknown>;
   332→}): Promise<BrainItem> {
   333→  const itemId = `item_${randomUUID().replace(/-/g, "").slice(0, 12)}`;
   334→  const now = new Date();
   335→  const itemType = input.itemType || "note";
   336→  const metadata = input.metadata ?? {};
   337→  const project = input.projectId ? await getProjectById(input.projectId) : null;
   338→
   339→  const created = await db.item.create({
   340→    data: {
   341→      id: itemId,
   342→      projectId: project?.id ?? null,
   343→      itemType,
   344→      title: input.title,
   345→      rawContent: input.rawContent,
   346→      structuredSummary: input.structuredSummary ?? "",
   347→      visualDescription: input.visualDescription ?? "",
   348→      filePath: vaultFilePath(project, itemType, input.title, now),
   349→      assetPath: input.assetPath ?? "",
   350→      sourceApp: input.sourceApp ?? "",
   351→      sourceUrl: input.sourceUrl ?? "",
   352→      metadata: JSON.stringify(metadata),
   353→      capturedAt: now,
   354→      // Vector sidecar write (sqlite-vec pattern) — every item is semantically indexed
   355→      vector: {
   356→        create: {
   357→          id: newVectorId(),
   358→          projectId: project?.id ?? null,
   359→          ...embedForStorage(
   360→            `${input.title}\n${input.rawContent}\n${input.structuredSummary ?? ""}\n${input.visualDescription ?? ""}`
   361→          ),
   362→        },
   363→      },
   364→    },
   365→    include: { vector: true },
   366→  });
   367→
   368→  if (project) {
   369→    await db.project.update({ where: { id: project.id }, data: { updatedAt: now } });
   370→  }
   371→
   372→  // v2: write the real Obsidian-compatible .md file to disk (best-effort, sqlite_store parity)
   373→  await writeVaultFile(
   374→    created.filePath,
   375→    buildItemMarkdown({
   376→      id: created.id,
   377→      itemType: created.itemType,
   378→      projectName: project?.name ?? "Inbox",
   379→      sourceApp: created.sourceApp,
   380→      sourceUrl: created.sourceUrl,
   381→      capturedAt: created.capturedAt.toISOString(),
   382→      title: created.title,
   383→      structuredSummary: created.structuredSummary,
   384→      visualDescription: created.visualDescription,
   385→      suggestedFix: typeof metadata.fix === "string" ? metadata.fix : undefined,
   386→      rawContent: created.rawContent,
   387→    })
   388→  );
   389→
   390→  return toItem(created);
   391→}

   393→// ==========================================
   394→// TWO-WAY OBSIDIAN DISK SYNC (v2, port of memory_compressor.sync_markdown_edits_from_disk)
   395→// If the user edited a tracked .md file directly on disk (Obsidian / VS Code),
   396→// sync the new content back into SQLite and re-embed the semantic vector.
   397→// ==========================================
   398→
   399→export async function syncMarkdownEditsFromDisk(): Promise<DiskSyncStats> {
   400→  const stats: DiskSyncStats = { files_checked: 0, sqlite_records_updated: 0, vectors_resynced: 0 };
   401→
   402→  const rows = await db.item.findMany({
   403→    where: { filePath: { not: "" } },
   404→    select: {
   405→      id: true,
   406→      title: true,
   407→      rawContent: true,
   408→      filePath: true,
   409→      structuredSummary: true,
   410→      visualDescription: true,
   411→      projectId: true,
   412→    },
   413→    take: 2000,
   414→    orderBy: { capturedAt: "desc" },
   415→  });
   416→
   417→  for (const r of rows) {
   418→    stats.files_checked += 1;
   419→    let diskText: string;
   420→    try {
   421→      diskText = await readFile(resolveVaultPath(r.filePath), "utf-8");
   422→    } catch {
   423→      continue; // file missing/unreadable — skip silently
   424→    }
   425→
   426→    // The editable payload lives after the "## Content\n" marker
   427→    const marker = "## Content\n";
   428→    let newRaw: string;
   429→    if (diskText.includes(marker)) {
   430→      newRaw = diskText.split(marker, 2)[1].trim();
   431→    } else {
   432→      newRaw = diskText.trim();
   433→    }
   434→
   435→    if (newRaw && newRaw !== r.rawContent.trim()) {
   436→      await db.item.update({ where: { id: r.id }, data: { rawContent: newRaw } });
   437→      stats.sqlite_records_updated += 1;
   438→
   439→      // Re-embed so FTS-equivalent fields AND semantic search stay in sync
   440→      await db.itemVector.upsert({
   441→        where: { itemId: r.id },
   442→        update: { ...embedForStorage(`${r.title}\n${newRaw}\n${r.structuredSummary}\n${r.visualDescription}`), indexedAt: new Date() },
   443→        create: {
   444→          id: newVectorId(),
   445→          itemId: r.id,
   446→          projectId: r.projectId,
   447→          ...embedForStorage(`${r.title}\n${newRaw}\n${r.structuredSummary}\n${r.visualDescription}`),
   448→        },
   449→      });
   450→      stats.vectors_resynced += 1;
   451→    }
   452→  }
   453→
   454→  return stats;
   455→}

   540→/**

   573→// ==========================================
   574→// SEARCH (FTS5 BM25 -> token OR match across indexed fields)
   575→// ==========================================
   576→
   577→export async function searchVault(
   578→  query: string,
   579→  projectId: string | null = null,
   580→  limit = 20
   581→): Promise<(BrainItem & { rank: number })[]> {
   582→  const cleanTokens = query
   583→    .split(/\s+/)
   584→    .map((w) => w.replace(/[^a-zA-Z0-9]/g, ""))
   585→    .filter((t) => t.length > 1);
   586→  if (cleanTokens.length === 0) return [];
   587→
   588→  // SQLite: case-insensitive contains across the FTS-indexed fields, OR-combined per token
   589→  const tokenFilter = cleanTokens.map((t) => ({
   590→    OR: [
   591→      { title: { contains: t } },
   592→      { rawContent: { contains: t } },
   593→      { structuredSummary: { contains: t } },
   594→      { visualDescription: { contains: t } },
   595→    ],
   596→  }));
   597→
   598→  const rows = await db.item.findMany({
   599→    where: { AND: [{ OR: tokenFilter }, ...(projectId ? [{ projectId }] : [])] },
   600→    orderBy: { capturedAt: "desc" },
   601→    take: limit * 3,
   602→  });
   603→
   604→  // BM25-ish ranking: more token hits across fields = better rank, tiebreak by recency
   605→  const scored = rows.map((row: ItemRow) => {
   606→    const item = toItem(row);
   607→    const haystack = `${item.title}\n${item.rawContent}\n${item.structuredSummary}\n${item.visualDescription}`.toLowerCase();
   608→    let hits = 0;
   609→    for (const t of cleanTokens) {
   610→      if (haystack.includes(t.toLowerCase())) hits += 1;
   611→    }
   612→    const ageHours = (Date.now() - new Date(item.capturedAt).getTime()) / 36e5;
   613→    const rank = hits * 10 + Math.max(0, 5 - ageHours / 24);
   614→    return { ...item, rank };
   615→  });
   616→
   617→  scored.sort((a, b) => b.rank - a.rank);
   618→  return scored.slice(0, limit);
   619→}

   621→export type SearchMode = "hybrid" | "semantic" | "fts";
   622→
   623→export interface SemanticSearchHit extends BrainItem {
   624→  rank: number;
   625→  /** Cosine similarity vs the query embedding (0..1). Present on vector-matched hits. */
   626→  similarity?: number;
   627→  /** Which ranking channels contributed: "vector" and/or "fts". */
   628→  modes: string[];
   629→}
   630→
   631→/**
   632→ * Hybrid semantic search (Sprint 3 upgrade):
   633→ *  - "semantic": pure cosine similarity over brain-hash-v1 embeddings
   634→ *  - "fts":      legacy BM25-like token ranking (searchVault)
   635→ *  - "hybrid":   0.6*vector + 0.25*token-overlap + 0.15*recency (default)
   636→ */
   637→export async function searchVaultHybrid(
   638→  query: string,
   639→  opts: {
   640→    projectId?: string | null;
   641→    limit?: number;
   642→    mode?: SearchMode;
   643→  } = {}
   644→): Promise<SemanticSearchHit[]> {
   645→  const { projectId = null, limit = 20, mode = "hybrid" } = opts;
   646→  if (!query.trim()) return [];
   647→
   648→  if (mode === "fts") {
   649→    const fts = await searchVault(query, projectId, limit);
   650→    return fts.map((r) => ({ ...r, modes: ["fts"] }));
   651→  }
   652→
   653→  await ensureVectorIndex();
   654→  const cleanTokens = query
   655→    .split(/\s+/)
   656→    .map((w) => w.replace(/[^a-zA-Z0-9]/g, ""))
   657→    .filter((t) => t.length > 1);
   658→  const queryVec = embedText(query);
   659→
   660→  const [vectorRows, tokenHits] = await Promise.all([
   661→    db.itemVector.findMany({
   662→      where: projectId ? { projectId } : {},
   663→      include: { item: true },
   664→      take: 3000,
   665→      orderBy: { indexedAt: "desc" },
   666→    }),
   667→    cleanTokens.length > 0
   668→      ? searchVault(query, projectId, limit * 3)
   669→      : Promise.resolve([]),
   670→  ]);
   671→
   672→  const tokenScore = new Map<string, number>();
   673→  if (cleanTokens.length > 0) {
   674→    for (const hit of tokenHits) {
   675→      const haystack =
   676→        `${hit.title}\n${hit.rawContent}\n${hit.structuredSummary}\n${hit.visualDescription}`.toLowerCase();
   677→      let hits = 0;
   678→      for (const t of cleanTokens) {
   679→        if (haystack.includes(t.toLowerCase())) hits += 1;
   680→      }
   681→      tokenScore.set(hit.id, hits / cleanTokens.length); // 0..1 coverage
   682→    }
   683→  }
   684→
   685→  const maxAgeHours = 24 * 14;
   686→  const scored: SemanticSearchHit[] = vectorRows.map((row) => {
   687→    const item = toItem(row.item);
   688→    const sim = Math.max(0, cosineSimilarity(queryVec, decodeVector(row.vector)));
   689→    const tScore = tokenScore.get(item.id) ?? 0;
   690→    const ageHours = (Date.now() - new Date(item.capturedAt).getTime()) / 36e5;
   691→    const recency = Math.max(0, 1 - ageHours / maxAgeHours);
   692→    const rank =
   693→      mode === "semantic"
   694→        ? sim
   695→        : 0.6 * sim + 0.25 * tScore + 0.15 * recency;
   696→    const modes: string[] = [];
   697→    if (sim > 0.01) modes.push("vector");
   698→    if (tScore > 0) modes.push("fts");
   699→    return { ...item, rank, similarity: sim, modes };
   700→  });
   701→
   702→  scored.sort((a, b) => b.rank - a.rank);
   703→  return scored.slice(0, limit);
   704→}

   736→// ==========================================
   737→// KNOWLEDGE HEALTH SCORE + GAP DETECTION (shared by details API & compressor)
   738→// ==========================================
   739→
   740→export interface ComputedHealth {
   741→  score: number;
   742→  gaps: { gap: string; detail: string; severity: "high" | "medium" | "low" }[];
   743→}
   744→
   745→/**
   746→ * Deterministic Knowledge Health scoring + gap detection.
   747→ * Base: 40 + 20 summary + 20 decisions + min(20, items*5); gaps subtract up to 20.
   748→ * Blueprint example: "40 screenshots of database errors but no Database Architecture doc" -> flagged.
   749→ */
   750→export function computeHealthAndGaps(
   751→  project: BrainProject,
   752→  items: BrainItem[],
   753→  decisions: BrainDecision[]
   754→): ComputedHealth {
   755→  const gaps: ComputedHealth["gaps"] = [];
   756→  let score =
   757→    40 +
   758→    (project.rollingSummary ? 20 : 0) +
   759→    (decisions.length > 0 ? 20 : 0) +
   760→    Math.min(20, items.length * 5);
   761→
   762→  // Gap 1 — unresolved error captures (blueprint: error surge without documented fix)
   763→  const unresolved = items.filter((it) => {
   764→    const isError = /error|exception|traceback|failed|cannot find|module not found/i.test(
   765→      `${it.title} ${it.rawContent}`
   766→    );
   767→    if (!isError) return false;
   768→    // metadata is already parsed into an object by toItem
   769→    return !it.metadata || !it.metadata.fix;
   770→  });
   771→  if (unresolved.length >= 3) {
   772→    gaps.push({
   773→      gap: "Unresolved error captures piling up",
   774→      detail: `${unresolved.length} error captures have no logged fix — log the fix or decision that resolved them`,
   775→      severity: "high",
   776→    });
   777→  }
   778→
   779→  // Gap 2 — visual-heavy, text-light memory
   780→  const shots = items.filter((i) => i.itemType === "screenshot").length;
   781→  const notes = items.filter((i) => i.itemType === "note").length;
   782→  if (shots >= 5 && notes === 0) {
   783→    gaps.push({
   784→      gap: "Visual-heavy memory without notes",
   785→      detail: `${shots} screenshots but zero written notes — screenshots are hard to recall without a summary note`,
   786→      severity: "medium",
   787→    });
   788→  }
   789→
   790→  // Gap 3 — captures without decisions
   791→  if (items.length >= 8 && decisions.length === 0) {
   792→    gaps.push({
   793→      gap: "Captures without logged decisions",
   794→      detail: `${items.length} captures yet no architectural decisions — capture choices with 'Decision: ... Reason: ...'`,
   795→      severity: "medium",
   796→    });
   797→  }
   798→
   799→  // Gap 4 — stale project memory
   800→  if (items.length > 0) {
   801→    const newest = Math.max(...items.map((i) => new Date(i.capturedAt).getTime()));
   802→    const staleDays = Math.floor((Date.now() - newest) / 864e5);
   803→    if (staleDays > 14) {
   804→      gaps.push({
   805→        gap: "Stale project memory",
   806→        detail: `No new captures in ${staleDays} days — project knowledge is going stale`,
   807→        severity: "low",
   808→      });
   809→    }
   810→  }
   811→
   812→  // Gap 5 — no compressed memory
   813→  if (!project.rollingSummary) {
   814→    gaps.push({
   815→      gap: "No compressed memory",
   816→      detail: "Run the Memory Compressor to synthesize Tier-1 rolling memory",
   817→      severity: "medium",
   818→    });
   819→  }
   820→
   821→  const penalty = gaps.reduce(
   822→    (acc, g) => acc + (g.severity === "high" ? 8 : g.severity === "medium" ? 4 : 2),
   823→    0
   824→  );
   825→  score = Math.max(0, Math.min(100, score - Math.min(20, penalty)));
   826→  return { score, gaps };
   827→}

   833→export async function getProjectDetails(projectId: string): Promise<ProjectDetails | { error: string }> {
   834→  const raw = await db.project.findUnique({ where: { id: projectId } });
   835→  if (!raw) return { error: "Project not found" };
   836→  const project = toProject(raw);
   837→
   838→  const [itemRows, decisionRows, edges] = await Promise.all([
   839→    db.item.findMany({ where: { projectId }, orderBy: { capturedAt: "desc" }, take: 50 }),
   840→    db.decision.findMany({ where: { projectId }, orderBy: { decidedAt: "desc" }, take: 30 }),
   841→    getProjectGraph(projectId),
   842→  ]);
   843→
   844→  const items = itemRows.map(toItem);
   845→  const decisions = decisionRows.map(toDecision);
   846→  const { score, gaps } = computeHealthAndGaps(project, items, decisions);
   847→
   848→  return {
   849→    project,
   850→    healthScore: score,
   851→    healthGaps: gaps,
   852→    lastCompressedAt: raw.lastCompressedAt ? raw.lastCompressedAt.toISOString() : null,
   853→    items,
   854→    decisions,
   855→    graphEdges: edges,
   856→  };
   857→}

   862→
   863→export async function buildContextInjectionPayload(
   864→  projectId: string,
   865→  activeQuery = ""
   866→): Promise<ContextPayload | { error: string }> {
   867→  const project = await getProjectById(projectId);
   868→  if (!project) return { error: `Project ${projectId} not found` };
   869→
   870→  const [decisionRows, recentRows, edges] = await Promise.all([
   871→    db.decision.findMany({ where: { projectId }, orderBy: { decidedAt: "desc" }, take: 6 }),
   872→    db.item.findMany({ where: { projectId }, orderBy: { capturedAt: "desc" }, take: 6 }),
   873→    getProjectGraph(projectId, 5),
   874→  ]);
   875→
   876→  const tier2 = decisionRows.map(toDecision).map((d) => ({
   877→    title: d.title,
   878→    decision: d.decision.length > 80 ? `${d.decision.slice(0, 80)}…` : d.decision,
   879→    reasoning: d.reasoning.length > 60 ? `${d.reasoning.slice(0, 60)}…` : d.reasoning,
   880→    outcome: d.outcome,
   881→    decidedAt: d.decidedAt,
   882→  }));
   883→
   884→  let tier3: ContextPayload["tier_3_recent_and_relevant_evidence"] = [];
   885→  if (activeQuery) {
   886→    const hits = await searchVault(activeQuery, projectId, 5);
   887→    tier3 = hits.map((h) => ({
   888→      title: h.title.slice(0, 70),
   889→      type: h.itemType,
   890→      summary: (h.structuredSummary || h.rawContent).slice(0, 140),
   891→      visual_context: h.visualDescription ? h.visualDescription.slice(0, 100) : undefined,
   892→      captured_at: h.capturedAt,
   893→    }));
   894→  }
   895→
   896→  if (tier3.length === 0) {
   897→    tier3 = recentRows.map((r) => {
   898→      const item = toItem(r);
   899→      return {
   900→        title: item.title,
   901→        type: item.itemType,
   902→        summary: item.structuredSummary.slice(0, 180),
   903→        captured_at: item.capturedAt,
   904→      };
   905→    });
   906→  }
   907→
   908→  return {
   909→    project_name: project.name,
   910→    tier_1_rolling_memory: project.rollingSummary || "Active development project.",
   911→    tier_2_decision_graph: tier2,
   912→    tier_3_recent_and_relevant_evidence: tier3,
   913→    // Slim edge projection keeps the JSON transport payload compact — the graph
   914→    // itself (full edges) is served by /api/projects/[id] for the Graph view.
   915→    knowledge_graph_edges: edges.map((e) => ({
   916→      source_node: e.sourceNode,
   917→      target_node: e.targetNode,
   918→      relation_type: e.relationType,
   919→    })),
   920→  };
   921→}

   923→/** Builds the anti-drift structured prompt wrapper (port of brain_api.py /context prompt rewriter). */
   924→export function buildStructuredPrompt(
   925→  payload: ContextPayload,
   926→  rawGoal = ""
   927→): string {
   928→  let prompt =
   929→    `# ROLE & GROUNDING CONSTRAINTS\n` +
   930→    `You are working on project **${payload.project_name}**.\n` +
   931→    `Follow the architectural decisions and evidence below strictly. Do NOT hallucinate unapproved libraries or revert past decisions.\n\n` +
   932→    `## TIER 1: COMPRESSED PROJECT MEMORY\n${payload.tier_1_rolling_memory}\n\n` +
   933→    `## TIER 2: ACTIVE ARCHITECTURAL DECISIONS (DO NOT VIOLATE)\n`;
   934→
   935→  for (const d of payload.tier_2_decision_graph) {
   936→    prompt += `- **${d.title}**: ${d.decision} (Reason: ${d.reasoning})\n`;
   937→  }
   938→
   939→  prompt += `\n## TIER 3: RELEVANT VAULT EVIDENCE & PAST FIXES\n`;
   940→  for (const ev of payload.tier_3_recent_and_relevant_evidence) {
   941→    prompt += `- [${ev.type.toUpperCase()}] **${ev.title}**: ${ev.summary}\n`;
   942→  }
   943→
   944→  if (rawGoal) {
   945→    prompt +=
   946→      `\n## CURRENT TASK / REWRITTEN EXECUTION SPEC\n` +
   947→      `**Objective:** ${rawGoal}\n` +
   948→      `**Required Output:** Provide modular, runnable code that respects Tier 1 & Tier 2 constraints and includes a quick verification step.\n`;
   949→  }
   950→
   951→  return prompt;
   952→}
