   110→   110→/**
   111→   111→ * Deterministic target positions for the guided layouts. Hierarchy: BFS layering
   112→   112→ * from root candidates (no incoming edges, else the highest out-degree node),
   113→   113→ * layered top-to-bottom. Radial: BFS rings around the highest-degree node.
   114→   114→ * Pure function — safe for the render path (no randomness).
   115→   115→ */
   116→   116→export function computeLayoutTargets(
   117→   117→  nodes: GraphViewerNode[],
   118→   118→  links: GraphViewerLink[],
   119→   119→  layout: GraphLayout,
   120→   120→): Map<string, { x: number; y: number }> | null {
   121→   121→  if (layout === "force" || nodes.length === 0) return null;
   122→   122→  const ids = nodes.map((n) => n.id);
   123→   123→  const outAdj = new Map<string, string[]>();
   124→   124→  const inDeg = new Map<string, number>();
   125→   125→  for (const id of ids) {
   126→   126→    outAdj.set(id, []);
   127→   127→    inDeg.set(id, 0);
   128→   128→  }
   129→   129→  for (const l of links) {
   130→   130→    if (!outAdj.has(l.source) || !outAdj.has(l.target) || l.source === l.target) continue;
   131→   131→    outAdj.get(l.source)!.push(l.target);
   132→   132→    inDeg.set(l.target, (inDeg.get(l.target) ?? 0) + 1);
   133→   133→  }
   134→   134→  const targets = new Map<string, { x: number; y: number }>();
   135→   135→
   136→   136→  if (layout === "hierarchy") {
   137→   137→    let roots = ids.filter((id) => (inDeg.get(id) ?? 0) === 0);
   138→   138→    if (roots.length === 0) {
   139→   139→      // Cyclic graph: seed from the most connected node.
   140→   140→      roots = [ids.reduce((a, b) => (outAdj.get(a)!.length >= outAdj.get(b)!.length ? a : b))];
   141→   141→    }
   142→   142→    const layerOf = new Map<string, number>();
   143→   143→    const queue = roots.map((id) => {
   144→   144→      layerOf.set(id, 0);
   145→   145→      return id;
   146→   146→    });
   147→   147→    for (let qi = 0; qi < queue.length; qi++) {
   148→   148→      const cur = queue[qi];
   149→   149→      const layer = layerOf.get(cur) ?? 0;
   150→   150→      for (const next of outAdj.get(cur) ?? []) {
   151→   151→        if (!layerOf.has(next)) {
   152→   152→          layerOf.set(next, layer + 1);
   153→   153→          queue.push(next);
   154→   154→        }
   155→   155→      }
   156→   156→    }
   157→   157→    const maxLayer = Math.max(0, ...[...layerOf.values()]);
   158→   158→    const buckets: string[][] = Array.from({ length: maxLayer + 1 }, () => []);
   159→   159→    for (const id of ids) buckets[layerOf.get(id) ?? maxLayer].push(id);
   160→   160→    buckets.forEach((bucket, layer) => {
   161→   161→      bucket.forEach((id, i) => {
   162→   162→        targets.set(id, {
   163→   163→          x: (i - (bucket.length - 1) / 2) * 120,
   164→   164→          y: layer * 110,
   165→   165→        });
   166→   166→      });
   167→   167→    });
   168→   168→    return targets;
   169→   169→  }
   170→   170→
   171→   171→  // radial — BFS rings around the most connected node
   172→   172→  const totalDeg = (id: string) => (outAdj.get(id)?.length ?? 0) + (inDeg.get(id) ?? 0);
   173→   173→  const center = ids.reduce((a, b) => (totalDeg(a) >= totalDeg(b) ? a : b));
   174→   174→  const ringOf = new Map<string, number>([[center, 0]]);
   175→   175→  const queue = [center];
   176→   176→  while (queue.length > 0) {
   177→   177→    const cur = queue.shift()!;
   178→   178→    const ring = ringOf.get(cur)!;
   179→   179→    for (const next of outAdj.get(cur) ?? []) {
   180→   180→      if (!ringOf.has(next)) {
   181→   181→        ringOf.set(next, ring + 1);
   182→   182→        queue.push(next);
   183→   183→      }
   184→   184→    }
   185→   185→    for (const id of inDeg.keys()) {
   186→   186→      // also walk incoming edges so directed-only backlinks land on a ring
   187→   187→      if (id === cur || ringOf.has(id)) continue;
   188→   188→      if ((outAdj.get(id) ?? []).includes(cur)) {
   189→   189→        ringOf.set(id, ring + 1);
   190→   190→        queue.push(id);
   191→   191→      }
   192→   192→    }
   193→   193→  }
   194→   194→  const maxRing = Math.max(0, ...[...ringOf.values()]);
   195→   195→  const rings: string[][] = Array.from({ length: maxRing + 1 }, () => []);
   196→   196→  for (const id of ids) rings[ringOf.get(id) ?? maxRing].push(id);
   197→   197→  rings.forEach((ring, d) => {
   198→   198→    const radius = d === 0 ? 0 : 90 + d * 95;
   199→   199→    ring.forEach((id, i) => {
   200→   200→      if (d === 0) {
   201→   201→        targets.set(id, { x: 0, y: 0 });
   202→   202→        return;
   203→   203→      }
   204→   204→      const angle = (i / ring.length) * Math.PI * 2 + d * 0.55;
   205→   205→      targets.set(id, { x: Math.cos(angle) * radius, y: Math.sin(angle) * radius });
   206→   206→    });
   207→   207→  });
   208→   208→  return targets;
   209→   209→}
   210→   210→
   211→   211→/**
   212→   212→ * One animation frame of the layout: pairwise repulsion (+soft collision), link
   213→   213→ * springs, then EITHER guided-target pull (hierarchy/radial) or gravity toward
   214→   214→ * the origin (force), followed by damped integration. Reads node data fresh from
   215→   215→ * propsRef every frame so data edits apply mid-animation.
   216→   216→ */
   217→   217→function runSimulation(ctx: SimContext) {
   218→   218→  const { nodes, links } = ctx.propsRef.current;
   219→   219→  const sims = ctx.sims;
   220→   220→  const list: Array<{ id: string; s: Sim }> = [];
   221→   221→  for (const n of nodes) {
   222→   222→    const s = sims.get(n.id);
   223→   223→    if (s) list.push({ id: n.id, s });
   224→   224→  }
   225→   225→
   226→   226→  // node repulsion (pairwise)
   227→   227→  for (let i = 0; i < list.length; i++) {
   228→   228→    const a = list[i].s;
   229→   229→    for (let j = i + 1; j < list.length; j++) {
   230→   230→      const b = list[j].s;
   231→   231→      const dx = a.x - b.x;
   232→   232→      const dy = a.y - b.y;
   233→   233→      const d2 = dx * dx + dy * dy + 0.01;
   234→   234→      const d = Math.sqrt(d2);
   235→   235→      let f = REPULSION / d2;
   236→   236→      if (d < MIN_GAP) f += (MIN_GAP - d) * 0.06; // soft collision
   237→   237→      const fx = (dx / d) * f;
   238→   238→      const fy = (dy / d) * f;
   239→   239→      a.vx += fx;
   240→   240→      a.vy += fy;
   241→   241→      b.vx -= fx;
   242→   242→      b.vy -= fy;
   243→   243→    }
   244→   244→  }
   245→   245→
   246→   246→  // link springs
   247→   247→  for (const l of links) {
   248→   248→    const a = sims.get(l.source);
   249→   249→    const b = sims.get(l.target);
   250→   250→    if (!a || !b) continue;
   251→   251→    const dx = b.x - a.x;
   252→   252→    const dy = b.y - a.y;
   253→   253→    const d = Math.hypot(dx, dy) || 1;
   254→   254→    const f = (d - LINK_REST) * LINK_STRENGTH;
   255→   255→    const fx = (dx / d) * f;
   256→   256→    const fy = (dy / d) * f;
   257→   257→    a.vx += fx;
   258→   258→    a.vy += fy;
   259→   259→    b.vx -= fx;
   260→   260→    b.vy -= fy;
   261→   261→  }
   262→   262→
   263→   263→  // gravity or guided-target pull + integrate
   264→   264→  const targets = ctx.targets.current;
   265→   265→  for (const { id, s } of list) {
   266→   266→    const target = targets?.get(id);
   267→   267→    if (target) {
   268→   268→      s.vx += (target.x - s.x) * 0.055;
   269→   269→      s.vy += (target.y - s.y) * 0.055;
   270→   270→    } else {
   271→   271→      s.vx += -s.x * GRAVITY;
   272→   272→      s.vy += -s.y * GRAVITY;
   273→   273→    }
   274→   274→    if (s.fx !== null && s.fy !== null) {
   275→   275→      s.x = s.fx;
   276→   276→      s.y = s.fy;
   277→   277→      s.vx = 0;
   278→   278→      s.vy = 0;
   279→   279→    } else {
   280→   280→      s.vx *= DAMPING;
   281→   281→      s.vy *= DAMPING;
   282→   282→      s.x += s.vx;
   283→   283→      s.y += s.vy;
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  // Publish a fresh render snapshot (keeps the React render path ref-free)
   288→   288→  const snapshot: Record<string, { x: number; y: number }> = {};
   289→   289→  for (const { id, s } of list) {
   290→   290→    snapshot[id] = { x: s.x, y: s.y };
   291→   291→  }
   292→   292→  ctx.setPositions(snapshot);
   293→   293→  ctx.alphaRef.current *= ALPHA_DECAY;
   294→   294→  ctx.rafRef.current =
   295→   295→    ctx.alphaRef.current > MIN_ALPHA ? requestAnimationFrame(() => runSimulation(ctx)) : null;
   296→   296→}

   298→   298→export const GraphViewer = ({
   299→   299→  nodes,
   300→   300→  links,
   301→   301→  onNodeClick,
   302→   302→  selectedId = null,
   303→   303→  height = 460,
   304→   304→  className = "",
   305→   305→  ariaLabel = "Interactive knowledge graph",
   306→   306→  loading = false,
   307→   307→}: GraphViewerProps) => {
   308→   308→  const { t } = useI18n();
   309→   309→  const svgRef = useRef<SVGSVGElement | null>(null);
   310→   310→
   311→   311→  // ---- layout mode (force | hierarchy | radial) ----
   312→   312→  const [layout, setLayout] = useState<GraphLayout>("force");
   313→   313→  const targetsRef = useRef<Map<string, { x: number; y: number }> | null>(null);
   314→   314→
   315→   315→  // ---- simulation state (refs: mutated by the rAF loop; painted via state) ----
   316→   316→  const simRef = useRef<Map<string, Sim>>(new Map());
   317→   317→  const alphaRef = useRef(0);
   318→   318→  const rafRef = useRef<number | null>(null);
   319→   319→  const propsRef = useRef({ nodes, links });
   320→   320→  // Sync latest graph data outside of render (react-hooks/refs-compliant)
   321→   321→  useEffect(() => {
   322→   322→    propsRef.current = { nodes, links };
   323→   323→  }, [nodes, links]);
   324→   324→
   325→   325→  // Render-side mirror of the live simulation — the loop publishes a fresh
   326→   326→  // snapshot each frame, keeping the render path ref-free (react-hooks/refs).
   327→   327→  const [positions, setPositions] = useState<Record<string, { x: number; y: number }>>({});
   328→   328→
   329→   329→  const [transform, setTransform] = useState<Transform>({ x: 460, y: 230, k: 1 });
   330→   330→  const [hoverId, setHoverId] = useState<string | null>(null);
   331→   331→  const dragRef = useRef<{ id: string; dx: number; dy: number; moved: boolean } | null>(null);
   332→   332→  const panRef = useRef<{ px: number; py: number; tx: number; ty: number; moved: boolean } | null>(null);
   333→   333→
   334→   334→  // ---- degree + radius (memoized) ----
   335→   335→  const degree = useMemo(() => {
   336→   336→    const d = new Map<string, number>();
   337→   337→    for (const l of links) {
   338→   338→      d.set(l.source, (d.get(l.source) ?? 0) + 1);
   339→   339→      d.set(l.target, (d.get(l.target) ?? 0) + 1);
   340→   340→    }
   341→   341→    return d;
   342→   342→  }, [links]);
   343→   343→
   344→   344→  const radiusOf = (id: string) => {
   345→   345→    const deg = degree.get(id) ?? 0;
   346→   346→    return 9 + Math.min(9, deg * 1.7);
   347→   347→  };
   348→   348→
   349→   349→  // ---- adjacency for hover highlighting ----
   350→   350→  const neighbors = useMemo(() => {
   351→   351→    const m = new Map<string, Set<string>>();
   352→   352→    const add = (a: string, b: string) => {
   353→   353→      if (!m.has(a)) m.set(a, new Set());
   354→   354→      m.get(a)!.add(b);
   355→   355→    };
   356→   356→    for (const l of links) {
   357→   357→      add(l.source, l.target);
   358→   358→      add(l.target, l.source);
   359→   359→    }
   360→   360→    return m;
   361→   361→  }, [links]);
   362→   362→
   363→   363→  // ---- deterministic initial layout (golden-angle spiral; pure render-safe) ----
   364→   364→  const ensureSims = useCallback(() => {
   365→   365→    const sims = simRef.current;
   366→   366→    propsRef.current.nodes.forEach((n, i) => {
   367→   367→      if (sims.has(n.id)) return;
   368→   368→      const angle = i * 2.399963; // golden angle
   369→   369→      const r = 60 + Math.sqrt(i + 1) * 46;
   370→   370→      sims.set(n.id, {
   371→   371→        x: Math.cos(angle) * r,
   372→   372→        y: Math.sin(angle) * r,
   373→   373→        vx: 0,
   374→   374→        vy: 0,
   375→   375→        fx: null,
   376→   376→        fy: null,
   377→   377→      });
   378→   378→    });
   379→   379→    for (const id of [...sims.keys()]) {
   380→   380→      if (!propsRef.current.nodes.some((n) => n.id === id)) sims.delete(id);
   381→   381→    }
   382→   382→  }, []);
   383→   383→
   384→   384→  // ---- force tick (module-scope loop; self-recursion lives outside React) ----
   385→   385→  const tick = useCallback(() => {
   386→   386→    runSimulation({ propsRef, sims: simRef.current, alphaRef, rafRef, setPositions, targets: targetsRef });
   387→   387→  }, [setPositions]);
   388→   388→
   389→   389→  const reheat = useCallback(
   390→   390→    (alpha: number) => {
   391→   391→      alphaRef.current = Math.max(alphaRef.current, alpha);
   392→   392→      if (rafRef.current === null) rafRef.current = requestAnimationFrame(tick);
   393→   393→    },
   394→   394→    [tick],
   395→   395→  );
   396→   396→
   397→   397→  // (re)initialize on data or layout change
   398→   398→  useEffect(() => {
   399→   399→    targetsRef.current = computeLayoutTargets(nodes, links, layout);
   400→   400→    ensureSims();
   401→   401→    reheat(1);
   402→   402→  }, [nodes, links, layout, ensureSims, reheat]);
   403→   403→
   404→   404→  // Stop the loop on unmount. rafRef MUST be reset to null: React StrictMode
   405→   405→  // runs effect -> cleanup -> effect in dev, and a stale non-null id would make
   406→   406→  // reheat() believe a frame is still pending, permanently killing the layout.
   407→   407→  useEffect(() => {
   408→   408→    return () => {
   409→   409→      if (rafRef.current !== null) {
   410→   410→        cancelAnimationFrame(rafRef.current);
   411→   411→        rafRef.current = null;
   412→   412→      }
   413→   413→    };
   414→   414→  }, []);
   415→   415→
   416→   416→  // ---- zoom (non-passive wheel listener for preventDefault) ----
   417→   417→  const zoomAt = useCallback((clientX: number, clientY: number, factor: number) => {
   418→   418→    const el = svgRef.current;
   419→   419→    if (!el) return;
   420→   420→    const rect = el.getBoundingClientRect();
   421→   421→    const px = clientX - rect.left;
   422→   422→    const py = clientY - rect.top;
   423→   423→    setTransform((tr) => {
   424→   424→      const k = clamp(tr.k * factor, 0.2, 3.2);
   425→   425→      return { k, x: px - ((px - tr.x) / tr.k) * k, y: py - ((py - tr.y) / tr.k) * k };
   426→   426→    });
   427→   427→  }, []);
   428→   428→
   429→   429→  useEffect(() => {
   430→   430→    const el = svgRef.current;
   431→   431→    if (!el) return;
   432→   432→    const onWheel = (e: WheelEvent) => {
   433→   433→      e.preventDefault();
   434→   434→      zoomAt(e.clientX, e.clientY, Math.exp(-e.deltaY * 0.0016));
   435→   435→    };
   436→   436→    el.addEventListener("wheel", onWheel, { passive: false });
   437→   437→    return () => el.removeEventListener("wheel", onWheel);
   438→   438→  }, [zoomAt]);
   439→   439→
   440→   440→  const fitView = useCallback(() => {
   441→   441→    const el = svgRef.current;
   442→   442→    if (!el) return;
   443→   443→    const rect = el.getBoundingClientRect();
   444→   444→    const sims = [...simRef.current.values()];
   445→   445→    if (sims.length === 0 || rect.width === 0) return;
   446→   446→    const minX = Math.min(...sims.map((s) => s.x));
   447→   447→    const maxX = Math.max(...sims.map((s) => s.x));
   448→   448→    const minY = Math.min(...sims.map((s) => s.y));
   449→   449→    const maxY = Math.max(...sims.map((s) => s.y));
   450→   450→    const bw = Math.max(maxX - minX, 80);
   451→   451→    const bh = Math.max(maxY - minY, 80);
   452→   452→    const k = clamp(Math.min(rect.width / bw, rect.height / bh) * 0.82, 0.25, 1.8);
   453→   453→    setTransform({
   454→   454→      k,
   455→   455→      x: rect.width / 2 - ((minX + maxX) / 2) * k,
   456→   456→      y: rect.height / 2 - ((minY + maxY) / 2) * k,
   457→   457→    });
   458→   458→  }, []);
   459→   459→
   460→   460→  const zoomBy = (factor: number) => {
   461→   461→    const el = svgRef.current;
   462→   462→    if (!el) return;
   463→   463→    const rect = el.getBoundingClientRect();
   464→   464→    zoomAt(rect.left + rect.width / 2, rect.top + rect.height / 2, factor);
   465→   465→  };
   466→   466→
   467→   467→  /** Blueprint "Zoom & Pan Controls": slider maps 20..320 percent onto zoom k. */
   468→   468→  const onSliderZoom = (values: number[]) => {
   469→   469→    const el = svgRef.current;
   470→   470→    if (!el) return;
   471→   471→    const rect = el.getBoundingClientRect();
   472→   472→    const desired = clamp(values[0] / 100, 0.2, 3.2);
   473→   473→    zoomAt(rect.left + rect.width / 2, rect.top + rect.height / 2, desired / transform.k);
   474→   474→  };
   475→   475→
   476→   476→  // ---- pointer interactions ----
   477→   477→  const toGraph = (clientX: number, clientY: number) => {
   478→   478→    const el = svgRef.current;
   479→   479→    const rect = el!.getBoundingClientRect();
   480→   480→    return {
   481→   481→      gx: (clientX - rect.left - transform.x) / transform.k,
   482→   482→      gy: (clientY - rect.top - transform.y) / transform.k,
   483→   483→    };
   484→   484→  };
   485→   485→
   486→   486→  const capturePointer = (e: React.PointerEvent, el: Element) => {
   487→   487→    try {
   488→   488→      el.setPointerCapture(e.pointerId);
   489→   489→    } catch {
   490→   490→      // Synthetic pointers (tests, some touch stacks) have no active pointer id —
   491→   491→      // capture is an optimization here, not a requirement.
   492→   492→    }
   493→   493→  };
   494→   494→
   495→   495→  const onPointerDownNode = (e: React.PointerEvent, id: string) => {
   496→   496→    e.stopPropagation();
   497→   497→    capturePointer(e, e.currentTarget as Element);
   498→   498→    const { gx, gy } = toGraph(e.clientX, e.clientY);
   499→   499→    const s = simRef.current.get(id);
   500→   500→    if (!s) return;
   501→   501→    dragRef.current = { id, dx: gx - s.x, dy: gy - s.y, moved: false };
   502→   502→    s.fx = gx - dragRef.current.dx;
   503→   503→    s.fy = gy - dragRef.current.dy;
   504→   504→    reheat(0.45);
   505→   505→  };
   506→   506→
   507→   507→  const onPointerDownBg = (e: React.PointerEvent) => {
   508→   508→    capturePointer(e, e.currentTarget as Element);
   509→   509→    panRef.current = { px: e.clientX, py: e.clientY, tx: transform.x, ty: transform.y, moved: false };
   510→   510→  };
   511→   511→
   512→   512→  const onPointerMove = (e: React.PointerEvent) => {
   513→   513→    if (dragRef.current) {
   514→   514→      const { gx, gy } = toGraph(e.clientX, e.clientY);
   515→   515→      const s = simRef.current.get(dragRef.current.id);
   516→   516→      if (s) {
   517→   517→        s.fx = gx - dragRef.current.dx;
   518→   518→        s.fy = gy - dragRef.current.dy;
   519→   519→        dragRef.current.moved = true;
   520→   520→        reheat(0.4);
   521→   521→      }
   522→   522→      return;
   523→   523→    }
   524→   524→    if (panRef.current) {
   525→   525→      const dx = e.clientX - panRef.current.px;
   526→   526→      const dy = e.clientY - panRef.current.py;
   527→   527→      if (Math.abs(dx) + Math.abs(dy) > 3) panRef.current.moved = true;
   528→   528→      setTransform((tr) => ({ ...tr, x: panRef.current!.tx + dx, y: panRef.current!.ty + dy }));
   529→   529→    }
   530→   530→  };
   531→   531→
   532→   532→  const onPointerUp = () => {
   533→   533→    if (dragRef.current) {
   534→   534→      const { id, moved } = dragRef.current;
   535→   535→      const s = simRef.current.get(id);
   536→   536→      if (s) {
   537→   537→        s.fx = null;
   538→   538→        s.fy = null;
   539→   539→      }
   540→   540→      if (!moved && onNodeClick) {
   541→   541→        const node = propsRef.current.nodes.find((n) => n.id === id);
   542→   542→        if (node) onNodeClick(node);
   543→   543→      }
   544→   544→      dragRef.current = null;
   545→   545→    }
   546→   546→    panRef.current = null;
   547→   547→  };
   548→   548→
   549→   549→  // ---- rendering ----
   550→   550→  const isDimmed = (id: string) =>
   551→   551→    hoverId !== null && id !== hoverId && !(neighbors.get(hoverId)?.has(id) ?? false);
   552→   552→
   553→   553→  return (
   554→   554→    <div className={`relative overflow-hidden rounded-xl border border-border bg-[var(--graph-canvas)] ${className}`}>
   555→   555→      <svg
   556→   556→        ref={svgRef}
   557→   557→        role="application"
   558→   558→        aria-label={ariaLabel}
   559→   559→        style={{ height, touchAction: "none" }}
   560→   560→        className="w-full cursor-grab active:cursor-grabbing"
   561→   561→        onPointerDown={onPointerDownBg}
   562→   562→        onPointerMove={onPointerMove}
   563→   563→        onPointerUp={onPointerUp}
   564→   564→        onPointerLeave={onPointerUp}
   565→   565→      >
   566→   566→        <defs>
   567→   567→          <pattern id="graph-grid" width="26" height="26" patternUnits="userSpaceOnUse">
   568→   568→            <circle cx="1" cy="1" r="1" fill="var(--graph-grid)" />
   569→   569→          </pattern>
   570→   570→          {RELATIONS.map((rel) => (
   571→   571→            <marker
   572→   572→              key={rel}
   573→   573→              id={`gv-arrow-${rel}`}
   574→   574→              viewBox="0 0 10 10"
   575→   575→              refX="9"
   576→   576→              refY="5"
   577→   577→              markerWidth="6"
   578→   578→              markerHeight="6"
   579→   579→              orient="auto-start-reverse"
   580→   580→            >
   581→   581→              <path d="M 0 0 L 10 5 L 0 10 z" fill={relationColor(rel)} />
   582→   582→            </marker>
   583→   583→          ))}
   584→   584→          <marker id="gv-arrow-default" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6" markerHeight="6" orient="auto-start-reverse">
   585→   585→            <path d="M 0 0 L 10 5 L 0 10 z" fill="var(--graph-edge)" />
   586→   586→          </marker>
   587→   587→        </defs>
   588→   588→        <rect width="100%" height="100%" fill="url(#graph-grid)" opacity={0.5} />
   589→   589→
   590→   590→        <g transform={`translate(${transform.x},${transform.y}) scale(${transform.k})`}>
   591→   591→          {/* edges */}
   592→   592→          {links.map((l, i) => {
   593→   593→            const a = positions[l.source];
   594→   594→            const b = positions[l.target];
   595→   595→            if (!a || !b) return null;
   596→   596→            const dx = b.x - a.x;
   597→   597→            const dy = b.y - a.y;
   598→   598→            const len = Math.hypot(dx, dy) || 1;
   599→   599→            const ox = (-dy / len) * len * 0.1;
   600→   600→            const oy = (dx / len) * len * 0.1;
   601→   601→            const cx = (a.x + b.x) / 2 + ox;
   602→   602→            const cy = (a.y + b.y) / 2 + oy;
   603→   603→            // pull the end back so the arrowhead meets the node ring
   604→   604→            const ex = cx - b.x;
   605→   605→            const ey = cy - b.y;
   606→   606→            const el = Math.hypot(ex, ey) || 1;
   607→   607→            const r2 = radiusOf(l.target);
   608→   608→            const endX = b.x + (ex / el) * (r2 + 5);
   609→   609→            const endY = b.y + (ey / el) * (r2 + 5);
   610→   610→            const rel = l.relation ?? "";
   611→   611→            const highlighted = hoverId !== null && (l.source === hoverId || l.target === hoverId);
   612→   612→            const relatedToSelection =
   613→   613→              selectedId !== null && (l.source === selectedId || l.target === selectedId);
   614→   614→            return (
   615→   615→              <path
   616→   616→                key={l.id ?? `${l.source}->${l.target}-${i}`}
   617→   617→                d={`M ${a.x} ${a.y} Q ${cx} ${cy} ${endX} ${endY}`}
   618→   618→                fill="none"
   619→   619→                stroke={rel && RELATIONS.includes(rel) ? relationColor(rel) : "var(--graph-edge)"}
   620→   620→                strokeWidth={highlighted || relatedToSelection ? 2 : 1.2}
   621→   621→                strokeOpacity={highlighted || relatedToSelection ? 0.95 : 0.45}
   622→   622→                markerEnd={rel && RELATIONS.includes(rel) ? `url(#gv-arrow-${rel})` : "url(#gv-arrow-default)"}
   623→   623→              />
   624→   624→            );
   625→   625→          })}
   626→   626→
   627→   627→          {/* nodes */}
   628→   628→          {nodes.map((n) => {
   629→   629→            const p = positions[n.id];
   630→   630→            if (!p) return null;
   631→   631→            const r = radiusOf(n.id);
   632→   632→            const kindColor = KIND_COLOR[n.kind ?? "concept"] ?? KIND_COLOR.concept;
   633→   633→            const dim = isDimmed(n.id);
   634→   634→            const selected = selectedId === n.id;
   635→   635→            return (
   636→   636→              <g
   637→   637→                key={n.id}
   638→   638→                opacity={dim ? 0.22 : 1}
   639→   639→                onPointerDown={(e) => onPointerDownNode(e, n.id)}
   640→   640→                onPointerEnter={() => !dragRef.current && !panRef.current && setHoverId(n.id)}
   641→   641→                onPointerLeave={() => setHoverId((h) => (h === n.id ? null : h))}
   642→   642→                style={{ cursor: "pointer" }}
   643→   643→              >
   644→   644→                {selected && (
   645→   645→                  <circle cx={p.x} cy={p.y} r={r + 5} fill="none" stroke="var(--primary)" strokeWidth={1.4} strokeOpacity={0.9} />
   646→   646→                )}
   647→   647→                <circle
   648→   648→                  cx={p.x}
   649→   649→                  cy={p.y}
   650→   650→                  r={r}
   651→   651→                  fill="var(--graph-node-fill)"
   652→   652→                  stroke={kindColor}
   653→   653→                  strokeWidth={selected || hoverId === n.id ? 2.4 : 1.6}
   654→   654→                />
   655→   655→                <circle cx={p.x} cy={p.y} r={r - 3.5} fill={kindColor} fillOpacity={0.18} />
   656→   656→                <text
   657→   657→                  x={p.x}
   658→   658→                  y={p.y + r + 12}
   659→   659→                  textAnchor="middle"
   660→   660→                  fontSize={10}
   661→   661→                  fontFamily="var(--font-geist-mono), monospace"
   662→   662→                  fill="var(--graph-node-text)"
   663→   663→                  opacity={0.92}
   664→   664→                  style={{ pointerEvents: "none", userSelect: "none" }}
   665→   665→                >
   666→   666→                  {truncate(n.label || n.id, LABEL_MAX)}
   667→   667→                </text>
   668→   668→              </g>
   669→   669→            );
   670→   670→          })}
   671→   671→        </g>
   672→   672→      </svg>
   673→   673→
   674→   674→      {/* controls */}
   675→   675→      <div className="absolute right-2.5 top-2.5 flex flex-col gap-1">
   676→   676→        <ControlButton label="Zoom in" onClick={() => zoomBy(1.25)}>
   677→   677→          <ZoomIn className="h-3.5 w-3.5" />
   678→   678→        </ControlButton>
   679→   679→        <ControlButton label="Zoom out" onClick={() => zoomBy(0.8)}>
   680→   680→          <ZoomOut className="h-3.5 w-3.5" />
   681→   681→        </ControlButton>
   682→   682→        <ControlButton label="Fit view" onClick={fitView}>
   683→   683→          <Maximize className="h-3.5 w-3.5" />
   684→   684→        </ControlButton>
   685→   685→        <ControlButton label="Re-run layout" onClick={() => reheat(1)}>
   686→   686→          <RefreshCw className="h-3.5 w-3.5" />
   687→   687→        </ControlButton>
   688→   688→      </div>
   689→   689→
   690→   690→      {/* layout selector + zoom slider (blueprint control set) */}
   691→   691→      <div className="absolute bottom-2 right-2.5 flex items-center gap-2 rounded-lg border border-border bg-card/90 px-2 py-1.5 backdrop-blur">
   692→   692→        <select
   693→   693→          value={layout}
   694→   694→          onChange={(e) => setLayout(e.target.value as GraphLayout)}
   695→   695→          aria-label="Graph layout algorithm"
   696→   696→          className="cursor-pointer rounded border border-border bg-background/70 px-1.5 py-0.5 text-[10px] text-foreground focus:outline-none"
   697→   697→        >
   698→   698→          <option value="force">{t("graph.layout.force")}</option>
   699→   699→          <option value="hierarchy">{t("graph.layout.hierarchy")}</option>
   700→   700→          <option value="radial">{t("graph.layout.radial")}</option>
   701→   701→        </select>
   702→   702→        <button
   703→   703→          type="button"
   704→   704→          onClick={() => zoomBy(0.8)}
   705→   705→          aria-label="Zoom out"
   706→   706→          className="text-muted-foreground transition-colors hover:text-foreground"
   707→   707→        >
   708→   708→          <ZoomOut className="h-3 w-3" />
   709→   709→        </button>
   710→   710→        <Slider
   711→   711→          value={[Math.round(transform.k * 100)]}
   712→   712→          min={20}
   713→   713→          max={320}
   714→   714→          step={5}
   715→   715→          onValueChange={onSliderZoom}
   716→   716→          aria-label="Zoom level"
   717→   717→          className="w-20"
   718→   718→        />
   719→   719→        <button
   720→   720→          type="button"
   721→   721→          onClick={() => zoomBy(1.25)}
   722→   722→          aria-label="Zoom in"
   723→   723→          className="text-muted-foreground transition-colors hover:text-foreground"
   724→   724→        >
   725→   725→          <ZoomIn className="h-3 w-3" />
   726→   726→        </button>
   727→   727→        <span className="w-8 text-right text-[10px] tabular-nums text-muted-foreground">
   728→   728→          {Math.round(transform.k * 100)}%
   729→   729→        </span>
   730→   730→      </div>
   731→   731→
   732→   732→      {loading && (
   733→   733→        <div
   734→   734→          className="absolute inset-0 z-10 flex items-center justify-center bg-card/80 backdrop-blur-sm"
   735→   735→          role="status"
   736→   736→          aria-label="Loading graph"
   737→   737→        >
   738→   738→          <Loader2 className="h-7 w-7 animate-spin text-primary" />
   739→   739→        </div>
   740→   740→      )}
   741→   741→
   742→   742→      <div className="pointer-events-none absolute bottom-2 left-3 text-[10px] text-muted-foreground/70">
   743→   743→        drag nodes · scroll to zoom · drag canvas to pan · click a node to inspect
   744→   744→      </div>
   745→   745→    </div>
   746→   746→  );
   747→   747→};
