   209→   209→export async function upsertProject(input: {
   210→   210→  name: string;
   211→   211→  rootPath?: string;
   212→   212→  keywords?: string[];
   213→   213→  gitRemotes?: string[];
   214→   214→  rollingSummary?: string;
   215→   215→}): Promise<BrainProject> {
   216→   216→  const slug = slugify(input.name);
   217→   217→  const keywords = input.keywords ?? [];
   218→   218→  const gitRemotes = input.gitRemotes ?? [];
   219→   219→
   220→   220→  const existing = await db.project.findUnique({ where: { slug } });
   221→   221→  let project: BrainProject;
   222→   222→
   223→   223→  if (existing) {
   224→   224→    const updated = await db.project.update({
   225→   225→      where: { id: existing.id },
   226→   226→      data: {
   227→   227→        name: input.name,
   228→   228→        rootPath: input.rootPath ?? existing.rootPath,
   229→   229→        keywords: JSON.stringify(keywords),
   230→   230→        gitRemotes: JSON.stringify(gitRemotes),
   231→   231→        rollingSummary: input.rollingSummary || existing.rollingSummary,
   232→   232→        updatedAt: new Date(),
   233→   233→      },
   234→   234→    });
   235→   235→    project = toProject(updated);
   236→   236→  } else {
   237→   237→    const created = await db.project.create({
   238→   238→      data: {
   239→   239→        id: `proj_${randomUUID().replace(/-/g, "").slice(0, 10)}`,
   240→   240→        name: input.name,
   241→   241→        slug,
   242→   242→        rootPath: input.rootPath ?? "",
   243→   243→        keywords: JSON.stringify(keywords),
   244→   244→        gitRemotes: JSON.stringify(gitRemotes),
   245→   245→        rollingSummary: input.rollingSummary ?? "",
   246→   246→      },
   247→   247→    });
   248→   248→    project = toProject(created);
   249→   249→  }
   250→   250→
   251→   251→  // v2: maintain the human-readable _project_summary.md on disk (sqlite_store.py parity:
   252→   252→  // writes when a summary exists or the file is missing)
   253→   253→  const effectiveSummary = project.rollingSummary;
   254→   254→  const summaryRelPath = `ProjectBrainVault/Projects/${project.slug}/_project_summary.md`;
   255→   255→  const summaryAbs = resolveVaultPath(summaryRelPath);
   256→   256→  let summaryExists = true;
   257→   257→  try {
   258→   258→    await readFile(summaryAbs, "utf-8");
   259→   259→  } catch {
   260→   260→    summaryExists = false;
   261→   261→  }
   262→   262→  if (effectiveSummary || !summaryExists) {
   263→   263→    const now = new Date().toISOString();
   264→   264→    await writeVaultFile(
   265→   265→      summaryRelPath,
   266→   266→      `# ${project.name} — Project Brain Summary\n\n` +
   267→   267→        `**Root Path:** \`${project.rootPath}\`\n` +
   268→   268→        `**Keywords:** ${project.keywords.join(", ")}\n` +
   269→   269→        `**Last Updated:** ${now}\n\n` +
   270→   270→        `## Rolling Memory\n${effectiveSummary || "No summary compressed yet."}\n`
   271→   271→    );
   272→   272→  }
   273→   273→
   274→   274→  return project;
   275→   275→}

   287→   287→/** Resolves proj_xxx id, slug, or display name -> project id (port of mcp resolve_project_id). */
   288→   288→export async function resolveProjectId(identifier: string): Promise<string | null> {
   289→   289→  if (!identifier) return null;
   290→   290→  const projects = await getAllProjects();
   291→   291→  for (const p of projects) {
   292→   292→    if (p.id === identifier || p.slug === identifier || p.name === identifier) return p.id;
   293→   293→  }
   294→   294→  const lower = identifier.toLowerCase();
   295→   295→  for (const p of projects) {
   296→   296→    if (p.name.toLowerCase() === lower || p.slug === lower) return p.id;
   297→   297→  }
   298→   298→  return null;
   299→   299→}

   305→   305→function vaultFilePath(project: BrainProject | null, itemType: string, title: string, now: Date): string {
   306→   306→  const stamp = `${now.getUTCFullYear()}${String(now.getUTCMonth() + 1).padStart(2, "0")}${String(
   307→   307→    now.getUTCDate()
   308→   308→  ).padStart(2, "0")}_${String(now.getUTCHours()).padStart(2, "0")}${String(now.getUTCMinutes()).padStart(
   309→   309→    2,
   310→   310→    "0"
   311→   311→  )}${String(now.getUTCSeconds()).padStart(2, "0")}`;
   312→   312→  const fileSlug = slugify(title).slice(0, 50);
   313→   313→  const filename = `${stamp}_${fileSlug}.md`;
   314→   314→  if (project) {
   315→   315→    const subfolder = ["screenshot", "clip", "terminal"].includes(itemType) ? "captures" : "notes";
   316→   316→    return `ProjectBrainVault/Projects/${project.slug}/${subfolder}/${filename}`;
   317→   317→  }
   318→   318→  return `ProjectBrainVault/Inbox/${filename}`;
   319→   319→}

   321→   321→export async function saveItem(input: {
   322→   322→  title: string;
   323→   323→  rawContent: string;
   324→   324→  itemType?: string;
   325→   325→  projectId?: string | null;
   326→   326→  structuredSummary?: string;
   327→   327→  visualDescription?: string;
   328→   328→  assetPath?: string;
   329→   329→  sourceApp?: string;
   330→   330→  sourceUrl?: string;
   331→   331→  metadata?: Record<string, unknown>;
   332→   332→}): Promise<BrainItem> {
   333→   333→  const itemId = `item_${randomUUID().replace(/-/g, "").slice(0, 12)}`;
   334→   334→  const now = new Date();
   335→   335→  const itemType = input.itemType || "note";
   336→   336→  const metadata = input.metadata ?? {};
   337→   337→  const project = input.projectId ? await getProjectById(input.projectId) : null;
   338→   338→
   339→   339→  const created = await db.item.create({
   340→   340→    data: {
   341→   341→      id: itemId,
   342→   342→      projectId: project?.id ?? null,
   343→   343→      itemType,
   344→   344→      title: input.title,
   345→   345→      rawContent: input.rawContent,
   346→   346→      structuredSummary: input.structuredSummary ?? "",
   347→   347→      visualDescription: input.visualDescription ?? "",
   348→   348→      filePath: vaultFilePath(project, itemType, input.title, now),
   349→   349→      assetPath: input.assetPath ?? "",
   350→   350→      sourceApp: input.sourceApp ?? "",
   351→   351→      sourceUrl: input.sourceUrl ?? "",
   352→   352→      metadata: JSON.stringify(metadata),
   353→   353→      capturedAt: now,
   354→   354→      // Vector sidecar write (sqlite-vec pattern) — every item is semantically indexed
   355→   355→      vector: {
   356→   356→        create: {
   357→   357→          id: newVectorId(),
   358→   358→          projectId: project?.id ?? null,
   359→   359→          ...embedForStorage(
   360→   360→            `${input.title}\n${input.rawContent}\n${input.structuredSummary ?? ""}\n${input.visualDescription ?? ""}`
   361→   361→          ),
   362→   362→        },
   363→   363→      },
   364→   364→    },
   365→   365→    include: { vector: true },
   366→   366→  });
   367→   367→
   368→   368→  if (project) {
   369→   369→    await db.project.update({ where: { id: project.id }, data: { updatedAt: now } });
   370→   370→  }
   371→   371→
   372→   372→  // v2: write the real Obsidian-compatible .md file to disk (best-effort, sqlite_store parity)
   373→   373→  await writeVaultFile(
   374→   374→    created.filePath,
   375→   375→    buildItemMarkdown({
   376→   376→      id: created.id,
   377→   377→      itemType: created.itemType,
   378→   378→      projectName: project?.name ?? "Inbox",
   379→   379→      sourceApp: created.sourceApp,
   380→   380→      sourceUrl: created.sourceUrl,
   381→   381→      capturedAt: created.capturedAt.toISOString(),
   382→   382→      title: created.title,
   383→   383→      structuredSummary: created.structuredSummary,
   384→   384→      visualDescription: created.visualDescription,
   385→   385→      suggestedFix: typeof metadata.fix === "string" ? metadata.fix : undefined,
   386→   386→      rawContent: created.rawContent,
   387→   387→    })
   388→   388→  );
   389→   389→
   390→   390→  return toItem(created);
   391→   391→}

   394→   394→// TWO-WAY OBSIDIAN DISK SYNC (v2, port of memory_compressor.sync_markdown_edits_from_disk)
   395→   395→// If the user edited a tracked .md file directly on disk (Obsidian / VS Code),
   396→   396→// sync the new content back into SQLite and re-embed the semantic vector.
   397→   397→// ==========================================
   398→   398→
   399→   399→export async function syncMarkdownEditsFromDisk(): Promise<DiskSyncStats> {
   400→   400→  const stats: DiskSyncStats = { files_checked: 0, sqlite_records_updated: 0, vectors_resynced: 0 };
   401→   401→
   402→   402→  const rows = await db.item.findMany({
   403→   403→    where: { filePath: { not: "" } },
   404→   404→    select: {
   405→   405→      id: true,
   406→   406→      title: true,
   407→   407→      rawContent: true,
   408→   408→      filePath: true,
   409→   409→      structuredSummary: true,
   410→   410→      visualDescription: true,
   411→   411→      projectId: true,
   412→   412→    },
   413→   413→    take: 2000,
   414→   414→    orderBy: { capturedAt: "desc" },
   415→   415→  });
   416→   416→
   417→   417→  for (const r of rows) {
   418→   418→    stats.files_checked += 1;
   419→   419→    let diskText: string;
   420→   420→    try {
   421→   421→      diskText = await readFile(resolveVaultPath(r.filePath), "utf-8");
   422→   422→    } catch {
   423→   423→      continue; // file missing/unreadable — skip silently
   424→   424→    }
   425→   425→
   426→   426→    // The editable payload lives after the "## Content\n" marker
   427→   427→    const marker = "## Content\n";
   428→   428→    let newRaw: string;
   429→   429→    if (diskText.includes(marker)) {
   430→   430→      newRaw = diskText.split(marker, 2)[1].trim();
   431→   431→    } else {
   432→   432→      newRaw = diskText.trim();
   433→   433→    }
   434→   434→
   435→   435→    if (newRaw && newRaw !== r.rawContent.trim()) {
   436→   436→      await db.item.update({ where: { id: r.id }, data: { rawContent: newRaw } });
   437→   437→      stats.sqlite_records_updated += 1;
   438→   438→
   439→   439→      // Re-embed so FTS-equivalent fields AND semantic search stay in sync
   440→   440→      await db.itemVector.upsert({
   441→   441→        where: { itemId: r.id },
   442→   442→        update: { ...embedForStorage(`${r.title}\n${newRaw}\n${r.structuredSummary}\n${r.visualDescription}`), indexedAt: new Date() },
   443→   443→        create: {
   444→   444→          id: newVectorId(),
   445→   445→          itemId: r.id,
   446→   446→          projectId: r.projectId,
   447→   447→          ...embedForStorage(`${r.title}\n${newRaw}\n${r.structuredSummary}\n${r.visualDescription}`),
   448→   448→        },
   449→   449→      });
   450→   450→      stats.vectors_resynced += 1;
   451→   451→    }
   452→   452→  }
   453→   453→
   454→   454→  return stats;
   455→   455→}

   573→   573→// ==========================================
   574→   574→// SEARCH (FTS5 BM25 -> token OR match across indexed fields)
   575→   575→// ==========================================
   576→   576→
   577→   577→export async function searchVault(
   578→   578→  query: string,
   579→   579→  projectId: string | null = null,
   580→   580→  limit = 20
   581→   581→): Promise<(BrainItem & { rank: number })[]> {
   582→   582→  const cleanTokens = query
   583→   583→    .split(/\s+/)
   584→   584→    .map((w) => w.replace(/[^a-zA-Z0-9]/g, ""))
   585→   585→    .filter((t) => t.length > 1);
   586→   586→  if (cleanTokens.length === 0) return [];
   587→   587→
   588→   588→  // SQLite: case-insensitive contains across the FTS-indexed fields, OR-combined per token
   589→   589→  const tokenFilter = cleanTokens.map((t) => ({
   590→   590→    OR: [
   591→   591→      { title: { contains: t } },
   592→   592→      { rawContent: { contains: t } },
   593→   593→      { structuredSummary: { contains: t } },
   594→   594→      { visualDescription: { contains: t } },
   595→   595→    ],
   596→   596→  }));
   597→   597→
   598→   598→  const rows = await db.item.findMany({
   599→   599→    where: { AND: [{ OR: tokenFilter }, ...(projectId ? [{ projectId }] : [])] },
   600→   600→    orderBy: { capturedAt: "desc" },
   601→   601→    take: limit * 3,
   602→   602→  });
   603→   603→
   604→   604→  // BM25-ish ranking: more token hits across fields = better rank, tiebreak by recency
   605→   605→  const scored = rows.map((row: ItemRow) => {
   606→   606→    const item = toItem(row);
   607→   607→    const haystack = `${item.title}\n${item.rawContent}\n${item.structuredSummary}\n${item.visualDescription}`.toLowerCase();
   608→   608→    let hits = 0;
   609→   609→    for (const t of cleanTokens) {
   610→   610→      if (haystack.includes(t.toLowerCase())) hits += 1;
   611→   611→    }
   612→   612→    const ageHours = (Date.now() - new Date(item.capturedAt).getTime()) / 36e5;
   613→   613→    const rank = hits * 10 + Math.max(0, 5 - ageHours / 24);
   614→   614→    return { ...item, rank };
   615→   615→  });
   616→   616→
   617→   617→  scored.sort((a, b) => b.rank - a.rank);
   618→   618→  return scored.slice(0, limit);
   619→   619→}
   620→   620→
   621→   621→export type SearchMode = "hybrid" | "semantic" | "fts";
   622→   622→
   623→   623→export interface SemanticSearchHit extends BrainItem {
   624→   624→  rank: number;
   625→   625→  /** Cosine similarity vs the query embedding (0..1). Present on vector-matched hits. */
   626→   626→  similarity?: number;
   627→   627→  /** Which ranking channels contributed: "vector" and/or "fts". */
   628→   628→  modes: string[];
   629→   629→}
   630→   630→
   631→   631→/**
   632→   632→ * Hybrid semantic search (Sprint 3 upgrade):
   633→   633→ *  - "semantic": pure cosine similarity over brain-hash-v1 embeddings
   634→   634→ *  - "fts":      legacy BM25-like token ranking (searchVault)
   635→   635→ *  - "hybrid":   0.6*vector + 0.25*token-overlap + 0.15*recency (default)
   636→   636→ */
   637→   637→export async function searchVaultHybrid(
   638→   638→  query: string,
   639→   639→  opts: {
   640→   640→    projectId?: string | null;
   641→   641→    limit?: number;
   642→   642→    mode?: SearchMode;
   643→   643→  } = {}
   644→   644→): Promise<SemanticSearchHit[]> {
   645→   645→  const { projectId = null, limit = 20, mode = "hybrid" } = opts;
   646→   646→  if (!query.trim()) return [];
   647→   647→
   648→   648→  if (mode === "fts") {
   649→   649→    const fts = await searchVault(query, projectId, limit);
   650→   650→    return fts.map((r) => ({ ...r, modes: ["fts"] }));
   651→   651→  }
   652→   652→
   653→   653→  await ensureVectorIndex();
   654→   654→  const cleanTokens = query
   655→   655→    .split(/\s+/)
   656→   656→    .map((w) => w.replace(/[^a-zA-Z0-9]/g, ""))
   657→   657→    .filter((t) => t.length > 1);
   658→   658→  const queryVec = embedText(query);
   659→   659→
   660→   660→  const [vectorRows, tokenHits] = await Promise.all([
   661→   661→    db.itemVector.findMany({
   662→   662→      where: projectId ? { projectId } : {},
   663→   663→      include: { item: true },
   664→   664→      take: 3000,
   665→   665→      orderBy: { indexedAt: "desc" },
   666→   666→    }),
   667→   667→    cleanTokens.length > 0
   668→   668→      ? searchVault(query, projectId, limit * 3)
   669→   669→      : Promise.resolve([]),
   670→   670→  ]);
   671→   671→
   672→   672→  const tokenScore = new Map<string, number>();
   673→   673→  if (cleanTokens.length > 0) {
   674→   674→    for (const hit of tokenHits) {
   675→   675→      const haystack =
   676→   676→        `${hit.title}\n${hit.rawContent}\n${hit.structuredSummary}\n${hit.visualDescription}`.toLowerCase();
   677→   677→      let hits = 0;
   678→   678→      for (const t of cleanTokens) {
   679→   679→        if (haystack.includes(t.toLowerCase())) hits += 1;
   680→   680→      }
   681→   681→      tokenScore.set(hit.id, hits / cleanTokens.length); // 0..1 coverage
   682→   682→    }
   683→   683→  }
   684→   684→
   685→   685→  const maxAgeHours = 24 * 14;
   686→   686→  const scored: SemanticSearchHit[] = vectorRows.map((row) => {
   687→   687→    const item = toItem(row.item);
   688→   688→    const sim = Math.max(0, cosineSimilarity(queryVec, decodeVector(row.vector)));
   689→   689→    const tScore = tokenScore.get(item.id) ?? 0;
   690→   690→    const ageHours = (Date.now() - new Date(item.capturedAt).getTime()) / 36e5;
   691→   691→    const recency = Math.max(0, 1 - ageHours / maxAgeHours);
   692→   692→    const rank =
   693→   693→      mode === "semantic"
   694→   694→        ? sim
   695→   695→        : 0.6 * sim + 0.25 * tScore + 0.15 * recency;
   696→   696→    const modes: string[] = [];
   697→   697→    if (sim > 0.01) modes.push("vector");
   698→   698→    if (tScore > 0) modes.push("fts");
   699→   699→    return { ...item, rank, similarity: sim, modes };
   700→   700→  });
   701→   701→
   702→   702→  scored.sort((a, b) => b.rank - a.rank);
   703→   703→  return scored.slice(0, limit);
   704→   704→}

   737→   737→// KNOWLEDGE HEALTH SCORE + GAP DETECTION (shared by details API & compressor)
   738→   738→// ==========================================
   739→   739→
   740→   740→export interface ComputedHealth {
   741→   741→  score: number;
   742→   742→  gaps: { gap: string; detail: string; severity: "high" | "medium" | "low" }[];
   743→   743→}
   744→   744→
   745→   745→/**
   746→   746→ * Deterministic Knowledge Health scoring + gap detection.
   747→   747→ * Base: 40 + 20 summary + 20 decisions + min(20, items*5); gaps subtract up to 20.
   748→   748→ * Blueprint example: "40 screenshots of database errors but no Database Architecture doc" -> flagged.
   749→   749→ */
   750→   750→export function computeHealthAndGaps(
   751→   751→  project: BrainProject,
   752→   752→  items: BrainItem[],
   753→   753→  decisions: BrainDecision[]
   754→   754→): ComputedHealth {
   755→   755→  const gaps: ComputedHealth["gaps"] = [];
   756→   756→  let score =
   757→   757→    40 +
   758→   758→    (project.rollingSummary ? 20 : 0) +
   759→   759→    (decisions.length > 0 ? 20 : 0) +
   760→   760→    Math.min(20, items.length * 5);
   761→   761→
   762→   762→  // Gap 1 — unresolved error captures (blueprint: error surge without documented fix)
   763→   763→  const unresolved = items.filter((it) => {
   764→   764→    const isError = /error|exception|traceback|failed|cannot find|module not found/i.test(
   765→   765→      `${it.title} ${it.rawContent}`
   766→   766→    );
   767→   767→    if (!isError) return false;
   768→   768→    // metadata is already parsed into an object by toItem
   769→   769→    return !it.metadata || !it.metadata.fix;
   770→   770→  });
   771→   771→  if (unresolved.length >= 3) {
   772→   772→    gaps.push({
   773→   773→      gap: "Unresolved error captures piling up",
   774→   774→      detail: `${unresolved.length} error captures have no logged fix — log the fix or decision that resolved them`,
   775→   775→      severity: "high",
   776→   776→    });
   777→   777→  }
   778→   778→
   779→   779→  // Gap 2 — visual-heavy, text-light memory
   780→   780→  const shots = items.filter((i) => i.itemType === "screenshot").length;
   781→   781→  const notes = items.filter((i) => i.itemType === "note").length;
   782→   782→  if (shots >= 5 && notes === 0) {
   783→   783→    gaps.push({
   784→   784→      gap: "Visual-heavy memory without notes",
   785→   785→      detail: `${shots} screenshots but zero written notes — screenshots are hard to recall without a summary note`,
   786→   786→      severity: "medium",
   787→   787→    });
   788→   788→  }
   789→   789→
   790→   790→  // Gap 3 — captures without decisions
   791→   791→  if (items.length >= 8 && decisions.length === 0) {
   792→   792→    gaps.push({
   793→   793→      gap: "Captures without logged decisions",
   794→   794→      detail: `${items.length} captures yet no architectural decisions — capture choices with 'Decision: ... Reason: ...'`,
   795→   795→      severity: "medium",
   796→   796→    });
   797→   797→  }
   798→   798→
   799→   799→  // Gap 4 — stale project memory
   800→   800→  if (items.length > 0) {
   801→   801→    const newest = Math.max(...items.map((i) => new Date(i.capturedAt).getTime()));
   802→   802→    const staleDays = Math.floor((Date.now() - newest) / 864e5);
   803→   803→    if (staleDays > 14) {
   804→   804→      gaps.push({
   805→   805→        gap: "Stale project memory",
   806→   806→        detail: `No new captures in ${staleDays} days — project knowledge is going stale`,
   807→   807→        severity: "low",
   808→   808→      });
   809→   809→    }
   810→   810→  }
   811→   811→
   812→   812→  // Gap 5 — no compressed memory
   813→   813→  if (!project.rollingSummary) {
   814→   814→    gaps.push({
   815→   815→      gap: "No compressed memory",
   816→   816→      detail: "Run the Memory Compressor to synthesize Tier-1 rolling memory",
   817→   817→      severity: "medium",
   818→   818→    });
   819→   819→  }
   820→   820→
   821→   821→  const penalty = gaps.reduce(
   822→   822→    (acc, g) => acc + (g.severity === "high" ? 8 : g.severity === "medium" ? 4 : 2),
   823→   823→    0
   824→   824→  );
   825→   825→  score = Math.max(0, Math.min(100, score - Math.min(20, penalty)));
   826→   826→  return { score, gaps };
   827→   827→}
